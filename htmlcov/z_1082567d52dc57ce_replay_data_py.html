1"""app.replay_data: part of AtlasSession (a mixin; see app/session.py): the timelapse and "Colour by". 

2 

3The page plays the replay; the session works out its plan (app.replay.make_plan): the step each screenshot, map, 

4icon and landmark appears at. While the replay is open those steps go into the vector tiles (a "step" property the 

5page filters on). The ground not revealed yet is covered by fog: a grid of the step each spot's ground is revealed at, 

6which the page draws every frame. "Colour by" needs every screenshot's outline, session and age rank, and the "times

33 """Return a colour spread (View tab) as 256 CSS colours.

35 Args: 

36 name: A COLOUR_SPREADS name (default plasma). 

37 

38 Returns:

77 def _age_info(self): 

78 """Return every screenshot's time order, session, age rank (0 oldest .. 1 newest) and time. 

79 

80 Returns: 

81 (order, sessions, rank, times), or None without screenshots. 

82 """ 

83 frames = self.atlas.frames if self.atlas else [] 

84 if not frames: 

85 return None 

86 key = (id(frames), len(frames)) 

87 c = getattr(self, "_age_cache", None) 

88 if c is None or c[0] != key: 

89 times = [R.capture_time(f) for f in frames] 

90 order = R.time_order(times) 

91 rank = np.zeros(len(frames))

93 c = self._age_cache = (key, (order, R.sessions(times), rank, times)) 

94 return c[1]

97 def frames_payload(self) -> dict: 

98 """Return what "Colour by" and the replay draw per screenshot, plus the colours. 

99 

100 Returns: 

101 {"frames": [{"name", "corners", "session", "rank", "time"}], "order" (oldest first), "ramp" (256 colours), 

102 "sessions" (colours), "legend": [{"label", "colour"}], "oldest", "newest", "alpha", "heat_top"}. 

103 """ 

104 info = self._age_info() 

105 style = self.ui_settings()["style"] 

106 out = { 

107 "frames": [], 

108 "order": [], 

109 "ramp": spread_colours(style.get("colour_spread")), 

110 "sessions": [bgr_hex(c) for c in REPLAY_SESSION_COLS], 

111 "legend": [], 

112 "oldest": "", 

113 "newest": "", 

114 "alpha": REPLAY_TINT_ALPHA, 

115 "heat_top": self.heat_top() if info else 1, 

116 } 

117 if info is None: 

118 return out 

119 order, sess, rank, times = info 

120 fp = self.D.get("footprints") or [] 

121 for i, f in enumerate(self.atlas.frames): 

122 corners = fp[i][0].tolist() if i < len(fp) else [] 

123 out["frames"].append( 

124 { 

125 "name": f["name"], 

126 "corners": corners, 

127 "session": int(sess[i]), 

128 "rank": float(rank[i]), 

129 "time": times[i], 

130 } # fmt: skip 

131 ) 

132 out["order"] = [int(i) for i in order] 

133 starts: dict[int, float | None] = {} 

134 for i in order: 

135 starts.setdefault(sess[i], times[i]) 

136 out["legend"] = [ 

137 {"label": R.session_label(t), "colour": out["sessions"][s % len(REPLAY_SESSION_COLS)]} 

138 for s, t in sorted(starts.items()) 

139 ][-6:] 

140 timed = sorted(t for t in times if t is not None) 

141 out["oldest"] = R.session_label(timed[0]) if timed else "oldest" 

142 out["newest"] = R.session_label(timed[-1]) if timed else "newest" 

143 return out

145 def _heat_grid(self, shown: np.ndarray): 

146 """Count how many of the shown screenshots saw each cell of a grid over the atlas. 

147 

148 Args: 

149 shown: bool per screenshot. 

150 

151 Returns: 

152 (counts uint16 (h, w), (x0, y0, cell)). 

153 """ 

154 x0, y0, x1, y1 = self.atlas_bounds() 

155 cell = max(x1 - x0, y1 - y0, 1.0) / HEAT_RES 

156 count = np.zeros((int((y1 - y0) / cell) + 2, int((x1 - x0) / cell) + 2), np.uint16) 

157 H, W = count.shape 

158 fp = self.D.get("footprints") or [] 

159 for i in np.nonzero(shown[: len(fp)])[0]: 

160 q = ((np.asarray(fp[i][0], float) - [x0, y0]) / cell).astype(np.int32) 

161 a, b = np.maximum(q.min(axis=0), 0), np.minimum(q.max(axis=0) + 1, [W, H]) 

162 if (b <= a).any(): 

163 continue 

164 sub = np.zeros((b[1] - a[1], b[0] - a[0]), np.uint8) 

165 cv2.fillConvexPoly(sub, q - a, 1) # one area at a time (overlapping polygons in one call cancel) 

166 count[a[1] : b[1], a[0] : b[0]] += sub 

167 return count, (x0, y0, cell)

169 def heat_top(self) -> int: 

170 """Return the most times any spot was seen (over all screenshots; fixed so replay colours don't shift). 

171 

172 Returns: 

173 The count (at least 1). 

174 """ 

175 key = (self.model_version, len(self.atlas.frames) if self.atlas else 0) 

176 c = getattr(self, "_heat_top", None) 

177 if c is None or c[0] != key: 

178 n = len(self.atlas.frames) if self.atlas else 0 

179 count, _o = self._heat_grid(np.ones(n, bool)) 

180 c = self._heat_top = (key, max(1, int(count.max()) if count.size else 1)) 

181 return c[1]

183 def heat_png(self, upto: float | None = None) -> tuple[bytes, list[float]] | None: 

184 """Make the "times seen" picture: each spot coloured by how many screenshots saw it, see-through elsewhere. 

185 

186 Args: 

187 upto: Only screenshots revealed before this replay step (None: all). 

188 

189 Returns: 

190 (PNG bytes, [x0, y0, x1, y1] it covers in atlas units), or None without screenshots. 

191 """ 

192 if not self.atlas or not self.atlas.frames: 

193 return None 

194 n = len(self.atlas.frames) 

195 shown = np.ones(n, bool) 

196 plan = getattr(self, "_plan", None) 

197 if upto is not None and plan is not None and len(plan.step_of) == n: 

198 shown = np.asarray(plan.step_of) < upto 

199 count, (x0, y0, cell) = self._heat_grid(shown) 

200 top = self.heat_top() 

201 cmap = COLOUR_SPREADS.get(self.ui_settings()["style"].get("colour_spread"), cv2.COLORMAP_PLASMA) 

202 ramp = cv2.applyColorMap(np.arange(256, dtype=np.uint8).reshape(-1, 1), cmap)[:, 0] 

203 idx = np.clip((count.astype(np.float32) - 1) / max(1, top - 1) * 255, 0, 255).astype(np.uint8) 

204 alpha = ((count > 0) * 255).astype(np.uint8) 

205 ok, buf = cv2.imencode(".png", np.dstack([ramp[idx], alpha])) 

206 H, W = count.shape 

207 return (buf.tobytes(), [x0, y0, x0 + W * cell, y0 + H * cell]) if ok else None

269 def steps(items, pts): 

270 """Steps of maps / icons: where the sweep reaches them, else when the first screenshot whose area holds 

271 them comes in (with the ground under them). By the screenshots that detected them, 3081 of 13697 map 

272 nodes of the crawl atlas came in long after their ground (a dim map node is often found only later).""" 

273 pts = np.asarray(pts, float).reshape(-1, 2) 

274 if plan.sweep is not None: 

275 return plan.steps_at(pts) 

276 s = R.point_steps(pts, fp, plan.step_of, default=-1) 

277 miss = np.nonzero(s < 0)[0] # outside every screenshot's area: by the screenshots that saw it 

278 if len(miss): 

279 by = R.item_steps([items[i].get("frames", []) for i in miss], frames, plan.step_of, default=late) 

280 s[miss] = by 

281 return s

283 V = self.V 

284 node_pts = self.D.get("node_pts", np.zeros((0, 2))) 

285 icon_pts = np.array([[i["x"], i["y"]] for i in V["icons"]], float).reshape(-1, 2)

308 def replay_payload(self) -> dict: 

309 """Return what the page's replay player needs. 

310 

311 Returns: 

312 {"settings", "n_steps", "kind", "area" (reveal by area), "step_of" (per screenshot), "ticks", 

313 "step_time" (labels), "step_session", "real", "frame" (sweep frame) and "step_key" (grow camera), 

314 "maps_at" (maps revealed by each step, cumulative), "bounds"}. 

315 """ 

316 plan = self._plan 

317 st = self.replay_settings() 

318 if plan is None: 

319 return {"settings": st} 

320 n = plan.n_steps 

321 first: dict[int, int] = {} 

322 for i in plan.order: 

323 first.setdefault(min(n - 1, int(plan.step_of[i])), plan.session_of[i]) 

324 step_session, last = [], 0 

325 for s in range(n): 

326 last = first.get(s, last) 

327 step_session.append(int(last)) 

328 alive = np.array([nd["status"] != "removed" for nd in self.V["nodes"]], bool) 

329 ns = np.asarray(self._replay_steps["node"], float) 

330 maps_at = [] 

331 if len(ns): 

332 hist = ( 

333 np.bincount(np.clip(np.floor(ns[alive]).astype(int), 0, n - 1), minlength=n) 

334 if alive.any() 

335 else np.zeros(n) 

336 ) 

337 maps_at = np.cumsum(hist).astype(int).tolist() 

338 real = R.real_span(plan.times) 

339 took = "" 

340 if real > 0: 

341 h, m = divmod(int(real // 60), 60) 

342 took = f"{h} h {m} min" if h else (f"{m} min" if m else f"{int(real)} s") 

343 return { 

344 "settings": st, 

345 "token": getattr(self, "replay_token", 0), 

346 "n_steps": n, 

347 "kind": plan.kind, 

348 "area": plan.sweep is not None, 

349 "step_of": [float(v) for v in plan.step_of], 

350 "ticks": [int(t) for t in plan.ticks], 

351 "step_time": [R.session_label(t) if t is not None else "" for t in plan.step_time], 

352 "step_session": step_session, 

353 "real": took, 

354 "real_seconds": real, 

355 "frame": None 

356 if plan.frame is None 

357 else { 

358 "start": [float(v) for v in plan.frame[0]], 

359 "right": float(plan.frame[1]), 

360 "top": float(plan.frame[2]), 

361 "span": float(plan.frame[3]), 

362 }, # fmt: skip 

363 "step_key": None if plan.step_key is None else [float(v) for v in plan.step_key], 

364 "maps_at": maps_at, 

365 "bounds": list(self.atlas_bounds()), 

366 }

368 def replay_fog(self) -> dict: 

369 """Return the fog field: for each cell of a grid over the atlas, the replay step its ground is revealed at. 

370 

371 The page draws the fog from it every frame (hidden while the replay is before that step, fading in over a soft 

372 edge when revealing by area), so the sweep moves smoothly; fog bands of polygons snapped in steps (user). 

373 

374 Returns: 

375 {"field": {"w", "h", "top" (the step a value of 65534 means), "data" (base64 little-endian uint16 per cell, 

376 row by row; 65535 = no ground there), "corners" ([lng, lat] of the grid's top left, top right, bottom 

377 right, bottom left)}} or {"field": None} without a plan. 

378 """ 

379 plan = getattr(self, "_plan", None) 

380 if plan is None or self.grid is None: 

381 return {"field": None} 

382 if getattr(self, "_fog", None) is not None: 

383 return self._fog 

384 n = max(1, plan.n_steps) 

385 x0, y0, x1, y1 = self.atlas_bounds() 

386 cell = max(x1 - x0, y1 - y0, 1.0) / REPLAY_FOG_RES 

387 W, H = int(math.ceil((x1 - x0) / cell)) + 1, int(math.ceil((y1 - y0) / cell)) + 1 

388 if plan.sweep is not None: 

389 g, (gx, gy, gc), top = plan.sweep.grid, plan.sweep.origin, plan.sweep.top 

390 A = np.array([[cell / gc, 0, (x0 - gx) / gc], [0, cell / gc, (y0 - gy) / gc]], np.float32) 

391 key = cv2.warpAffine(g, A, (W, H), flags=cv2.INTER_LINEAR | cv2.WARP_INVERSE_MAP, borderValue=float(top)) 

392 steps = key.astype(np.float64) / max(float(top), 1e-9) * n 

393 ground = np.ones((H, W), bool) 

394 fp = self.D.get("footprints") or [] 

395 if fp: # only where screenshots show ground 

396 ground[:] = False 

397 for poly, _bb in fp: 

398 m = np.zeros((H, W), np.uint8) 

399 cv2.fillConvexPoly(m, ((np.asarray(poly, float) - [x0, y0]) / cell).round().astype(np.int32), 1) 

400 ground |= m > 0 

401 else: 

402 fp = self.D.get("footprints") or [] 

403 steps = np.full((H, W), np.inf) 

404 for i, (poly, _bb) in enumerate(fp): 

405 m = np.zeros((H, W), np.uint8) 

406 cv2.fillConvexPoly(m, ((np.asarray(poly, float) - [x0, y0]) / cell).round().astype(np.int32), 1) 

407 np.minimum(steps, np.where(m > 0, float(plan.step_of[i]), np.inf), out=steps) 

408 ground = np.isfinite(steps) 

409 enc = np.where(ground, np.clip(np.nan_to_num(steps, posinf=n) / n * 65534, 0, 65534), 65535).astype("<u2") 

410 corners = [self.grid.lnglat(x0 + cx * cell, y0 + cy * cell) for cx, cy in ((0, 0), (W, 0), (W, H), (0, H))] 

411 fog = { 

412 "field": {"w": W, "h": H, "top": n, "data": base64.b64encode(enc.tobytes()).decode(), "corners": corners} 

413 } 

414 with self._lock: # kept only while its plan is still the replay's (a replan or atlas switch meanwhile: not) 

415 if getattr(self, "_plan", None) is plan: 

416 self._fog = fog 

417 return fog
