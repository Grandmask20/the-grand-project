170 def _send(self, status: int, body: bytes, ctype: str, extra: dict | None = None) -> None: 

171 """Send a whole response. 

172 

173 Args: 

174 status: HTTP status. 

175 body: The body. 

176 ctype: Content-Type. 

177 extra: More headers. 

178 """ 

179 if not self._read_body and int(self.headers.get("Content-Length") or 0) > 0: 

180 # a body nobody read (a refused request) would be taken for the next request on this connection 

181 self.close_connection = True 

182 self.send_response(status) 

183 self.send_header("Content-Type", ctype) 

184 self.send_header("Content-Length", str(len(body))) 

185 # the atlas the answer was made from when the handler knows it, else the open one (read again here, it could 

186 # stamp one atlas's state with the next one's generation: GPT audit) 

187 self.send_header(GEN_HEADER, str(self._resp_gen if self._resp_gen is not None else self.S.generation)) 

188 if self.close_connection: 

189 self.send_header("Connection", "close") 

190 for k, v in (extra or {}).items(): 

191 self.send_header(k, v) 

192 self.end_headers() 

193 if self.command != "HEAD": 

194 self.wfile.write(body)

258 def _dispatch(self, method: str) -> None: 

259 """Answer a request: the API, else a file of the page. 

260 

261 Args: 

262 method: The HTTP method. 

263 """ 

264 self._read_body = False # per request (the handler lives as long as its keep-alive connection) 

265 self._resp_gen = None # the atlas an answer was made from (_send's header), when the handler knows it 

266 if not self._host_ok(): 

267 self._json({"error": "wrong host"}, 403) 

268 return 

269 url = urlparse(self.path) 

270 path = unquote(url.path) 

271 if not path.startswith("/api/"): 

272 if method in ("GET", "HEAD"): 

273 self._static(path) 

274 else: 

275 self._json({"error": "not found"}, 404) 

276 return 

277 stale = self.headers.get(GEN_HEADER) 

278 if ( 

279 method not in ("GET", "HEAD") 

280 and stale is not None 

281 and stale != str(self.S.generation) 

282 and not path.startswith(GEN_FREE) 

283 ): 

284 # a change the page made for the atlas open before (a bookmark id, a lasso, a placement): not applied to 

285 # the one open now (user, CodeRabbit) 

286 self._json({"error": "the atlas changed", "stale": True}, 409) 

287 return 

288 if method not in ("GET", "HEAD") and self.headers.get(HEADER) != "1": 

289 self._json({"error": f"missing {HEADER} header"}, 403) 

290 return 

291 # the atlas this request was made for: its header, else the one open as it came in (a switch while it is 

292 # handled refuses its change: CodeRabbit) 

293 self._req_gen = int(stale) if stale is not None and stale.isdigit() else self.S.generation 

294 q = parse_qs(url.query) 

295 # a change is for its atlas: every edit it makes checks that atlas under the session lock, in the same step 

296 # (AtlasLock; checked only here before, a switch in between got the edit: GPT audit) 

297 changes = method not in ("GET", "HEAD") and not path.startswith(GEN_FREE) 

298 for m, rx, name in ROUTES: 

299 if m != ("GET" if method == "HEAD" else method): 

300 continue 

301 hit = rx.match(path) 

302 if hit: 

303 try: 

304 with self.S._lock.expect(self._req_gen) if changes else contextlib.nullcontext(): 

305 getattr(self, name)(q, **hit.groupdict()) 

306 except ApiError as e: 

307 self._json({"error": str(e)}, e.status) 

308 except StaleAtlas: 

309 self._json({"error": "the atlas changed", "stale": True}, 409) 

310 except ValueError as e: 

311 self._json({"error": str(e)}, 400) 

312 except ConnectionError: # the browser gave up on it (tiles while zooming): nothing to report 

313 pass 

314 except Exception as e: 

315 LOG.exception(f"{method} {path} failed") 

316 with contextlib.suppress(OSError): 

317 self._json({"error": f"{type(e).__name__}: {e}"}, 500) 

318 return 

319 self._json({"error": "not found"}, 404)

385 """The live event stream (Server-Sent Events).""" 

386 lq = self.S.events.listen() 

387 self.send_response(200) 

388 self.send_header("Content-Type", "text/event-stream")

394 self.wfile.write(b"retry: 1500\n\n") 

395 self.wfile.flush()

399 except queue.Empty:

641 @route("POST", "/api/upload") 

642 def post_upload(self, q) -> None: 

643 """Save a screenshot dropped on the page into the open atlas's folder (the raw file is the body).""" 

644 if not self.S.path: 

645 raise ApiError("open or create an atlas first") 

646 name = Path(unquote(self.headers.get("X-Filename") or "")).name 

647 if not name or Path(name).suffix.lower() not in core.IMAGE_EXT: 

648 raise ApiError("drop screenshot image files (.png, .jpg)") 

649 n = int(self.headers.get("Content-Length") or 0) 

650 if n <= 0 or n > UPLOAD_MAX: 

651 raise ApiError("the file is empty or too large", 413) 

652 gen = self._req_gen # (read before the body: it can take a while) 

653 data = self.rfile.read(n) 

654 self._read_body = True 

655 with self.S._lock: 

656 opened = self.S.opened 

657 if not self.S.owns(gen) or opened.path is None: # another atlas opened while it came in (CodeRabbit) 

658 raise StaleAtlas() 

659 dest = opened.path.parent / name # (the folder of the atlas it was checked for, from its own record) 

660 stem, k = dest.stem, 0 

661 while dest.exists() and dest.read_bytes() != data: # another picture by that name: a free name (CodeRabbit) 

662 k += 1 

663 dest = dest.with_name(f"{stem}_{int(time.time())}{f'_{k}' if k > 1 else ''}{dest.suffix}") 

664 if not dest.exists(): 

665 dest.write_bytes(data) 

666 self._json({"path": str(dest)})

1324def reuse_running(port: int, atlas: str | None, browser: bool) -> str: 

1325 """Hand this start over to the copy already running, when there is one of this version. 

1326 

1327 The same version: its page is opened, and an atlas asked for on the command line is opened there (it was 

1328 dropped: CodeRabbit). An older version (the folder was updated while it ran): it is asked to quit, so this 

1329 start runs the update (the old copy kept being reused: CodeRabbit). 

1330 

1331 Args: 

1332 port: The first port the app would take. 

1333 atlas: The atlas file asked for on the command line, or None. 

1334 browser: Open the page. 

1335 

1336 Returns: 

1337 HANDED (the running copy took over), START (none running, or the old one quit: start normally) or FAILED 

1338 (a hand-over that didn't work: reported, and no second copy is started, CodeRabbit). 

1339 """ 

1340 found = running_app(port) 

1341 if not found: 

1342 return START 

1343 url, version = found 

1344 if version != VERSION: 

1345 _say(f"An older copy (v{version}) is running: closing it to start v{VERSION}") 

1346 _local(url + "api/quit", {}) 

1347 t0 = time.monotonic() 

1348 old_port = int(url.rstrip("/").rsplit(":", 1)[1]) 

1349 while time.monotonic() - t0 < APP_QUIT_WAIT: 

1350 # gone only when nothing listens there any more: a slow answer while it saves was taken for gone, and 

1351 # the new copy started beside it (CodeRabbit) 

1352 if _stopped(APP_HOST, old_port): 

1353 return START 

1354 time.sleep(0.25) 

1355 # still there (saving or finishing work): two copies would write the same atlas 

1356 _say(f"The older copy at {url} is still closing: start the app again in a moment.") 

1357 return FAILED 

1358 _say(f"The Grand Project is already running: {url}") 

1359 if browser: 

1360 webbrowser.open(url) 

1361 if atlas and _local(url + "api/atlas/open", {"path": str(Path(atlas).resolve()), "background": True}) is None: 

1362 _say(f"The running copy couldn't open {atlas} (it may be adding screenshots): open it there when it's done.") 

1363 return FAILED 

1364 return HANDED

1367def take_app_lock(): 

1368 """Take this program folder's start lock (user_data/app.lock), held while the app runs. 

1369 

1370 The look for a running copy and starting one are two steps: two launches at the same moment both found none and 

1371 both started on the next port, writing the same atlas (CodeRabbit). The lock makes one of them wait. The system 

1372 lets go of it when the process ends, however it ends. 

1373 

1374 Returns: 

1375 The open lock file (keep it while running), or None when another copy holds it. 

1376 """ 

1377 path = rc_paths.USER_DATA / APP_LOCK_FILE 

1378 path.parent.mkdir(parents=True, exist_ok=True) 

1379 f = open(path, "a+b") # noqa: SIM115 (held open for the app's life) 

1380 try: 

1381 if os.name == "nt": 

1382 import msvcrt 

1383 

1384 f.seek(0) 

1385 msvcrt.locking(f.fileno(), msvcrt.LK_NBLCK, 1) 

1386 else: 

1387 import fcntl 

1388 

1389 fcntl.flock(f.fileno(), fcntl.LOCK_EX | fcntl.LOCK_NB) 

1390 except OSError: 

1391 f.close() 

1392 return None 

1393 return f
