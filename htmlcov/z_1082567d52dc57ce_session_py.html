1"""app.session: the open atlas and everything the app does with it. 

2 

3One AtlasSession lives in the server. It holds the open atlas, the model the map shows (view model: map nodes, 

4paths, icons, landmarks, with the biomes worked out), the map image, the background worker that adds screenshots, 

5auto-add, Fix seams, undo, bookmarks, hand edits and debug marks. The browser asks for things through the HTTP API 

6(app.server) and hears about changes through Events (log, progress, "model" / "terrain" versions, questions). 

7 

8Each part lives in its own file as a mixin: library (known atlases, folders), 

9mapimage (map image, Fix seams, tiles), adding (the worker: add, drop, retry, undo, re-check), watch (auto-add), 

10edits (landmarks, biomes, city types, locked strongholds), bookmarks, query (search, details, screenshots of a 

11spot), debug (debug marks), replay (timelapse plan, colour by).

175 def open_atlas(self, p, expect_gen: int | None = None) -> None: 

176 """Open an atlas file and show it. 

177 

178 Unsaved changes of the open atlas are saved first. An atlas with screenshots but no atlas-wide result 

179 (made before the check existed) is checked now. Auto-add follows the atlas. 

180 

181 Args: 

182 p: Path of the atlas .json file. 

183 expect_gen: Open it only while this is still the generation (the app's start opening the last atlas in 

184 the background: an atlas opened on the page meanwhile stays, CodeRabbit). None: always. 

185 

186 Raises: 

187 ValueError: When the file cannot be read as an atlas, or work is still running. 

188 """ 

189 p = Path(p) 

190 if expect_gen is not None and not self.owns(expect_gen): 

191 return 

192 if self.working(): 

193 raise ValueError("wait until the screenshots being added are done") 

194 try: 

195 atlas = core.Atlas.load(p) 

196 except Exception as e: 

197 LOG.exception(f"could not read atlas {p}") 

198 raise ValueError(f"could not read {p}: {e}") from e 

199 switching = self.path is None or p.resolve() != self.path.resolve() 

200 # the open atlas's background work stops first (auto-add restarts below; Fix seams' result would be for the 

201 # atlas object read before, also when the same file is opened again). Not for the app's start (expect_gen): 

202 # nothing of its own is open yet, so any work running belongs to an atlas opened on the page meanwhile, and 

203 # stopping it there cancelled that atlas's Fix seams (CodeRabbit) 

204 if expect_gen is None: 

205 self.stop_watch() 

206 if self.seams_running(): 

207 self.stop_seams() 

208 self.events.log("Fix seams stopped: another atlas was opened") 

209 self.save() 

210 name = p.stem if p.stem != "atlas" else p.parent.name 

211 with self._lock: 

212 if expect_gen is not None and not self.owns(expect_gen): # another atlas opened while this one loaded 

213 return 

214 if self.working(): # a job slipped in meanwhile 

215 self._resume_watch() # (stopped above for a switch that doesn't happen: CodeRabbit) 

216 raise ValueError("wait until the screenshots being added are done") 

217 if switching: 

218 self._forget_session() 

219 self._drop_stale_snapshots(p) 

220 self._reset_mapimage() # the other atlas's map image never shows for this one 

221 before, now = self.opened, self._switch(atlas, p) 

222 # in the same step as the switch: set after it, an older load finishing late made itself the atlas 

223 # opened at the next start (CodeRabbit) 

224 self.settings["last_atlas"] = str(p) 

225 if before.scope and before.scope != now.scope: 

226 self.events.log(f"switched to atlas {name}", before.scope) 

227 maps = len((atlas.final or {}).get("nodes", [])) 

228 self.events.log(f"opened atlas {name} ({p}): {len(atlas.frames)} screenshots, {maps} maps", now.scope) 

229 if atlas.frames and atlas.final is None: 

230 self.events.log("this atlas predates the atlas-wide check: running it now") 

231 try: 

232 with self._lock: 

233 atlas_finalize.finalize(atlas, log=self.events.log) 

234 atlas.save(p) 

235 except Exception as e: 

236 LOG.exception("atlas-wide check failed") 

237 self.events.log(f"atlas-wide check failed: {e}") 

238 self._after_load() 

239 missing = [f["name"] for f in atlas.frames if not Path(f["path"]).exists()] 

240 if missing: 

241 self.events.log(f"screenshots not found (terrain missing there): {', '.join(missing[:20])}" + 

242 (f" and {len(missing) - 20} more" if len(missing) > 20 else "")) # fmt: skip 

243 if self.settings.get("watch_on"): 

244 self.start_watch() # auto-add follows the atlas: its own folder, or the one picked for it

280 def _switch(self, atlas: core.Atlas | None, p: Path | None) -> Opened: 

281 """Make `atlas` (at `p`; None: none) the open one, as a new generation (call under the lock). 

282 

283 The only place the open atlas changes: its record (atlas, path, generation, log) is swapped whole, so work 

284 that took the record never mixes one atlas's parts with another's. Read field by field, a switch between 

285 the reads gave a job atlas A's screenshots with B's generation, or A's log lines in B's log (CodeRabbit, 

286 many times over). 

287 

288 The model (map nodes, paths, icons, screenshot outlines) starts empty and is rebuilt right after: anything 

289 asked for meanwhile came from the atlas open before (user: the old atlas's times-seen picture blinked on the 

290 new one). 

291 

292 Returns: 

293 The new record. 

294 """ 

295 scope = str(p.resolve()) if p is not None else "" 

296 self.opened = Opened(atlas, p, self.opened.gen + 1, scope) 

297 self.parts.reset(self.opened.gen) 

298 self.V = {"nodes": [], "paths": [], "icons": [], "landmarks": [], "final": None} 

299 self.D = {} 

300 self.model_version += 1 

301 self.events.set_scope(scope) 

302 return self.opened

324 def owns(self, job: Opened | int) -> bool: 

325 """Tell whether work started for an atlas may still change things (call under the lock). 

326

328 atlas's record (`opened`) when it starts, works on that record's atlas, logs into its log, and applies its 

329 result only while it is still the open one: work for another atlas was installed into the one opened 

330 meanwhile (user, CodeRabbit). 

331 

332 Args: 

333 job: The record the work took, or its generation. 

334 

335 Returns: 

336 True while that atlas is still the open one. 

337 """ 

338 return (job.gen if isinstance(job, Opened) else job) == self.opened.gen

389 def rebuild_model(self) -> None: 

390 """Rebuild the view model from the atlas and everything derived from it: icon rows, type counts, locked 

391 strongholds, city types, biomes, screenshot footprints. Publishes a "model" event with the new version."""

403 def _rebuild_model(self, gen: int) -> None: 

404 """The work of rebuild_model, beating the model part between its steps. 

405 

406 Args: 

407 gen: The generation it was started for. 

408 """ 

409 with self._lock: 

410 V = view_model(self.atlas) if self.atlas else {"nodes": [], "paths": [], "icons": [], "landmarks": []} 

411 self.parts.beat("model", gen) 

412 D: dict = {} 

413 counts: dict[str, int] = {} 

414 for ic in V["icons"]: 

415 counts[icon_label(ic)] = counts.get(icon_label(ic), 0) + 1 

416 D["icon_types"] = sorted(counts, key=lambda k: (k == "Unidentified badge", k)) 

417 try: 

418 found = atlas_cities.biomes(self.atlas, V, ground=self.ground()) if self.atlas else None 

419 self.parts.beat("model", gen) 

420 except Exception: 

421 LOG.exception("city types / biomes failed") 

422 self.events.log("city types / biomes failed (details in the log file)") 

423 found = None 

424 if found: 

425 V["landmarks"] = found["landmarks"] 

426 for k in ("potential", "near_city", "blocked", "icon_node", "city_type", "locked"): 

427 D[k] = found[k] 

428 else: 

429 D.update(potential={}, near_city={}, blocked={}, icon_node={}, city_type={}, locked=set()) 

430 for lm in V["landmarks"]: 

431 if not lm["status"].startswith("removed"): 

432 counts[lm["kind"]] = counts.get(lm["kind"], 0) + 1 

433 for _i, (t, _how) in D["city_type"].items(): 

434 counts[f"{t} city"] = counts.get(f"{t} city", 0) + 1 

435 for _v, (lab, st) in D["potential"].items(): 

436 if st: 

437 k = potential_kind(lab) 

438 counts[k] = counts.get(k, 0) + 1 

439 D["place_types"] = [ 

440 k 

441 for k in ( 

442 "City", 

443 "Ezomyte city", 

444 "Vaal city", 

445 "Faridun city", 

446 "Hall", 

447 "Tower", 

448 "Breach stronghold", 

449 *POTENTIAL_TYPES, 

450 ) # fmt: skip 

451 if k in counts 

452 ] 

453 D["counts"] = counts 

454 D["node_pts"] = np.array([[n["x"], n["y"]] for n in V["nodes"]], float).reshape(-1, 2) 

455 D["icon_pts"] = icon_rows(V["nodes"], V["icons"], D["icon_node"]) if V["icons"] else np.zeros((0, 2)) 

456 D["footprints"] = [core.footprint(self.atlas, f) for f in self.atlas.frames] if self.atlas else [] 

457 D["landmark_node"] = landmark_nodes(V["nodes"], V["landmarks"], D["locked"]) 

458 self.V, self.D = V, D 

459 self.model_version += 1 

460 # screenshots added while the replay is open: replanned in the same locked step (checked apart, a 

461 # close between the check and the replan made the rebuild fail after it said the model was ready, 

462 # CodeRabbit) 

463 if getattr(self, "_replay_on", False) and self.atlas and self.atlas.frames: 

464 self._replan()

562def settings_change_text(changes: dict, hidden_before: set) -> str: 

563 """Say a settings change in one log line: "Paths layer off, paths drawn by at least 35%, types shown: Breach". 

564 

565 Args: 

566 changes: The change the page sent (set_ui_settings). 

567 hidden_before: The hidden types before it (only what changed is named). 

568 

569 Returns: 

570 The line. 

571 """ 

572 parts = [] 

573 for k, v in changes.items(): 

574 if k == "hidden_types": 

575 now = set(v) 

576 shown, hidden = sorted(hidden_before - now), sorted(now - hidden_before) 

577 for word, names in (("types shown", shown), ("types hidden", hidden)): 

578 if names: 

579 parts.append( 

580 f"{word}: " + ", ".join(names[:8]) + (f" and {len(names) - 8} more" if len(names) > 8 else "") 

581 ) 

582 elif k == "layers" and isinstance(v, dict): 

583 parts += [f"{kk} layer {_say(vv)}" for kk, vv in v.items()] 

584 elif k == "style" and isinstance(v, dict) and v.get("reset"): 

585 parts.append("styles reset") 

586 elif isinstance(v, dict): 

587 for kk, vv in v.items(): 

588 if kk == "player": # (I'm here) 

589 parts.append( 

590 f"your position map #{vv['node']}" if isinstance(vv, dict) else "your position forgotten" 

591 ) 

592 continue 

593 share = kk == "path_freq" and isinstance(vv, (int, float)) and not isinstance(vv, bool) 

594 val = f"{round(100 * vv)}%" if share else _say(vv) 

595 parts.append(f"{_WORDS.get(kk, kk.replace('_', ' '))} {val}") 

596 else: 

597 parts.append(f"{_WORDS.get(k, k)} {_say(v)}") 

598 return ", ".join(parts) or "nothing"

643def landmark_nodes(nodes: list[dict], landmarks: list[dict], locked=()) -> dict[int, int]: 

644 """Find the map node each landmark marks: the nearest live map node within reach (a locked stronghold has none). 

645 

646 A click on a landmark's 3D model selects this map node (user: clicking a city's citadel didn't light up its map). 

647 

648 Args: 

649 nodes: The view model's map nodes ("x", "y", "status"). 

650 landmarks: The view model's landmarks ("x", "y", "kind", "status"). 

651 locked: Indexes of locked strongholds: they stand apart, not at a map node (one nearby was picked: CodeRabbit). 

652 

653 Returns: 

654 Landmark index -> map node index. 

655 """ 

656 live = [i for i, n in enumerate(nodes) if n.get("status") != "removed"] 

657 if not live or not landmarks: 

658 return {} 

659 from scipy.spatial import cKDTree 

660 

661 tree = cKDTree(np.array([[nodes[i]["x"], nodes[i]["y"]] for i in live], float)) 

662 out = {} 

663 for k, lm in enumerate(landmarks): 

664 if str(lm.get("status", "")).startswith("removed") or k in locked: 

665 continue 

666 reach = LANDMARK_NODE_REACH_STRONGHOLD if lm.get("kind") == "Breach stronghold" else LANDMARK_NODE_REACH 

667 d, j = tree.query([lm["x"], lm["y"]]) 

668 if d <= reach: 

669 out[k] = live[int(j)] 

670 return out
