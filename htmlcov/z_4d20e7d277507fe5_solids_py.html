1"""webmap.solids: the web map's 3D view: map nodes, towers, citadels, halls and strongholds as extruded shapes. 

2 

3The atlas knows where each structure stands and what it is (landmark kind, city type, locked or not), not its 

4outline, so each kind gets a simple model in its own style: a pillar per map node, a stone shaft for a tower, a 

5stepped pyramid for a Vaal citadel, a dark walled ring for an Ezomyte one, sand walls with a green-roofed hall for a 

6Faridun one, a hall with its orange beam, a crystal cluster for a Breach stronghold. Sizes are in atlas units (one 

7atlas unit is about one screenshot pixel); the app turns heights into the metres MapLibre extrudes by (app.mapimage). 

8"""

19def _ngon(x: float, y: float, r: float, n: int = 8, turn: float = 0.0) -> list[tuple[float, float]]: 

20 """Return a regular polygon around a point. 

21 

22 Args: 

23 x: Centre x (atlas units). 

24 y: Centre y. 

25 r: Radius. 

26 n: Number of corners (4: a square standing on its side when turn is 45 degrees). 

27 turn: Rotation in degrees. 

28 

29 Returns: 

30 The ring's corners. 

31 """ 

32 a0 = math.radians(turn) 

33 return [(x + r * math.cos(a0 + 2 * math.pi * k / n), y + r * math.sin(a0 + 2 * math.pi * k / n)) for k in range(n)]

36def _square(x: float, y: float, side: float) -> list[tuple[float, float]]: 

37 """Return an axis-aligned square around a point. 

38 

39 Args: 

40 x: Centre x. 

41 y: Centre y. 

42 side: Side length. 

43 

44 Returns: 

45 The ring's corners. 

46 """ 

47 h = side / 2 

48 return [(x - h, y - h), (x + h, y - h), (x + h, y + h), (x - h, y + h)]

51def _solid(rings, base: float, top: float, colour: str, **extra) -> dict: 

52 """Return one extruded shape. 

53 

54 Args: 

55 rings: Outer ring, then holes (atlas units). 

56 base: Bottom height (atlas units). 

57 top: Top height (atlas units). 

58 colour: CSS colour. 

59 **extra: More properties (e.g. node: the map node it stands for, so a click selects it). 

60 

61 Returns: 

62 {"rings", "base", "top", "colour", ...}. 

63 """ 

64 return {"rings": rings, "base": base, "top": top, "colour": colour, **extra}

67def map_pillar(x: float, y: float, icons: list[str], colour: str, node: int) -> list[dict]: 

68 """Return the pillar of a map node: taller for a boss map, in its icon colour. 

69 

70 Args: 

71 x: Map node x. 

72 y: Map node y. 

73 icons: Its icon names. 

74 colour: Its colour (first icon's, or its status colour). 

75 node: Its web id. 

76 

77 Returns: 

78 The shapes. 

79 """ 

80 K = WEBMAP_SOLID 

81 top = K["boss_map"] if any(i in _DEADLY for i in icons) else K["map"] 

82 return [ 

83 _solid([_ngon(x, y, K["map_r"])], 0, top, colour, node=node), 

84 _solid([_ngon(x, y, K["map_r"] * 0.55)], top, top + K["map_r"] * 0.35, "#e8e2cf", node=node), # light cap 

85 ]

88def structure(kind: str, x: float, y: float, city_type: str = "", locked: bool = False) -> list[dict]: 

89 """Return the model of a landmark. 

90 

91 Args: 

92 kind: Landmark kind ("City", "Hall", "Tower", "Breach stronghold", "Deadly boss map"). 

93 x: Its x (atlas units). 

94 y: Its y. 

95 city_type: "Vaal" / "Ezomyte" / "Faridun" for a City whose type is known, else "". 

96 locked: A locked Breach stronghold (hand icon). 

97 

98 Returns: 

99 The shapes (empty for an unknown kind). 

100 """ 

101 S = [] 

102 if kind == "Tower": 

103 S += [ 

104 _solid([_square(x, y, 34)], 0, 150, "#8d8f95"), # shaft 

105 _solid([_square(x, y, 46)], 150, 168, "#6f7178"), # crown 

106 _solid([_ngon(x, y, 7)], 168, 196, "#9fe8ff"), # the beacon 

107 ] 

108 elif kind == "City" and city_type == "Vaal": # stepped pyramid 

109 for k, side in enumerate((160, 124, 88, 52)): 

110 S.append( 

111 _solid([_square(x, y, side)], 28 * k, 28 * (k + 1), ("#b8955a", "#c29f62", "#cba86a", "#d4b273")[k]) 

112 ) 

113 S.append(_solid([_square(x, y, 26)], 112, 136, "#e2c27e")) # the temple on top 

114 elif kind == "City" and city_type == "Ezomyte": # dark walled ring with wall towers and a keep 

115 S.append(_solid([_ngon(x, y, 88, 24), list(reversed(_ngon(x, y, 74, 24)))], 0, 42, "#4a4a50")) 

116 S += [_solid([_ngon(px, py, 12)], 0, 60, "#55555c") for px, py in _ngon(x, y, 81, 6, 15)] 

117 S += [_solid([_square(x, y, 42)], 0, 78, "#5d5d64"), _solid([_square(x, y, 50)], 78, 86, "#3e3e44")] 

118 elif kind == "City" and city_type == "Faridun": # sand walls, corner towers, a green-roofed hall 

119 S.append(_solid([_square(x, y, 164), list(reversed(_square(x, y, 144)))], 0, 36, "#c9b07a")) 

120 S += [_solid([_ngon(px, py, 15)], 0, 56, "#a99060") for px, py in _ngon(x, y, 77 * math.sqrt(2), 4, 45)] 

121 S += [_solid([_square(x, y, 62)], 0, 40, "#d3bb86"), _solid([_square(x, y, 66)], 40, 52, "#3f8a5a")] 

122 elif kind == "City": # type not known: a plain keep 

123 S += [_solid([_square(x, y, 70)], 0, 66, "#7c7a78"), _solid([_square(x, y, 78)], 66, 74, "#5f5d5b")] 

124 elif kind == "Hall": # a hall, and the orange beam that marks it in the game 

125 S += [ 

126 _solid([_square(x, y, 72)], 0, 44, "#6d6258"), 

127 _solid([_square(x, y, 80)], 44, 52, "#544a42"), 

128 _solid([_ngon(x, y, 6)], 52, 420, "#ff8c1a"), 

129 ] 

130 elif kind == "Breach stronghold": # a cluster of crystals; a locked one is dim and low 

131 tall = (140, 95, 120, 80, 110, 70, 100) 

132 for k, (px, py) in enumerate([(x, y), *_ngon(x, y, 26, 6, 10)]): 

133 S.append(_solid([_ngon(px, py, 11 if k == 0 else 8, 5, 18 * k)], 0, tall[k] * (0.45 if locked else 1), 

134 "#5a2a66" if locked else "#b84fd6")) # fmt: skip 

135 # a Deadly boss map has no building in the game, only its red icon: no model (a red post stood on it: user) 

136 return S
