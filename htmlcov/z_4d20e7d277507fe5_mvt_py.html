1"""webmap.mvt: a small Mapbox Vector Tile (2.1) encoder: points, lines and polygons with simple properties. 

2 

3No dependency: a vector tile is a short protocol-buffers message (layers -> features -> geometry commands), written 

4here by hand. Coordinates are tile-local integers (0..extent; a little outside is allowed, see WEBMAP_MVT_BUFFER). 

5""" 

6 

7from __future__ import annotations 

8 

9import struct 

10 

11_POINT, _LINESTRING, _POLYGON = 1, 2, 3 

12_MOVE_TO, _LINE_TO, _CLOSE_PATH = 1, 2, 7

15def _varint(n: int) -> bytes: 

16 """Encode a non-negative integer as a protocol-buffers varint. 

17 

18 Args: 

19 n: The integer (>= 0). 

20 

21 Returns: 

22 Its varint bytes. 

23 """ 

24 out = bytearray() 

25 while True: 

26 b = n & 0x7F 

27 n >>= 7 

28 if n: 

29 out.append(b | 0x80) 

30 else: 

31 out.append(b) 

32 return bytes(out)

35def _zigzag(n: int) -> int: 

36 """Map a signed integer to the unsigned zigzag form (0, -1, 1, -2 -> 0, 1, 2, 3). 

37 

38 Args: 

39 n: The signed integer. 

40 

41 Returns: 

42 Its zigzag value. 

43 """ 

44 return (n << 1) ^ (n >> 63)

47def _key(field: int, wire: int) -> bytes: 

48 """Encode a field key. 

49 

50 Args: 

51 field: Field number. 

52 wire: Wire type (0 varint, 1 64-bit, 2 length-delimited, 5 32-bit). 

53 

54 Returns: 

55 The key bytes. 

56 """ 

57 return _varint((field << 3) | wire)

60def _bytes_field(field: int, data: bytes) -> bytes: 

61 """Encode a length-delimited field (string, sub-message or packed list). 

62 

63 Args: 

64 field: Field number. 

65 data: The field's bytes. 

66 

67 Returns: 

68 The encoded field. 

69 """ 

70 return _key(field, 2) + _varint(len(data)) + data

73def _packed(field: int, values: list[int]) -> bytes: 

74 """Encode a packed repeated uint32 field. 

75 

76 Args: 

77 field: Field number. 

78 values: Non-negative integers. 

79 

80 Returns: 

81 The encoded field (empty bytes for an empty list). 

82 """ 

83 if not values: 

84 return b"" 

85 return _bytes_field(field, b"".join(_varint(v) for v in values))

88def _value(v) -> bytes: 

89 """Encode a property value as an MVT Value message. 

90 

91 Args: 

92 v: str, bool, int or float (anything else is stored as its str()). 

93 

94 Returns: 

95 The Value message bytes. 

96 """ 

97 if isinstance(v, bool): 

98 return _key(7, 0) + _varint(int(v)) 

99 if isinstance(v, int): 

100 return _key(6, 0) + _varint(_zigzag(v)) if v < 0 else _key(5, 0) + _varint(v) 

101 if isinstance(v, float): 

102 return _key(3, 1) + struct.pack("<d", v) 

103 return _bytes_field(1, str(v).encode("utf-8"))

106def ring_area(ring: list[tuple[int, int]]) -> float: 

107 """Return a ring's signed area in tile coordinates (y down): positive for an outer ring as MVT wants it. 

108 

109 Args: 

110 ring: The ring's vertices (not repeating the first). 

111 

112 Returns: 

113 The signed area. 

114 """ 

115 return 0.5 * sum(x0 * y1 - x1 * y0 for (x0, y0), (x1, y1) in zip(ring, ring[1:] + ring[:1], strict=True))

118def _polygon(rings: list[list[tuple[int, int]]]) -> list[int]: 

119 """Encode a polygon's rings (outer first, then its holes) as MVT geometry commands. 

120 

121 Args: 

122 rings: Rings of 3+ integer vertices, not repeating the first; outer positive area, holes negative. 

123 

124 Returns: 

125 The command integers. 

126 """ 

127 out, cx, cy = [], 0, 0 

128 for ring in rings: 

129 for i, (x, y) in enumerate(ring): 

130 if i == 0: 

131 out.append(_MOVE_TO | (1 << 3)) 

132 elif i == 1: 

133 out.append(_LINE_TO | ((len(ring) - 1) << 3)) 

134 out += [_zigzag(x - cx), _zigzag(y - cy)] 

135 cx, cy = x, y 

136 out.append(_CLOSE_PATH | (1 << 3)) 

137 return out

140def _geometry(kind: int, coords: list[tuple[int, int]]) -> list[int]: 

141 """Encode one point or one line as MVT geometry commands (zigzag deltas from the previous position). 

142 

143 Args: 

144 kind: _POINT or _LINESTRING. 

145 coords: [(x, y)] tile-local integers (one point, or 2+ line vertices). 

146 

147 Returns: 

148 The command integers. 

149 """ 

150 out, cx, cy = [], 0, 0 

151 for i, (x, y) in enumerate(coords): 

152 if i == 0: 

153 out.append(_MOVE_TO | (1 << 3)) 

154 elif i == 1 and kind == _LINESTRING: 

155 out.append(_LINE_TO | ((len(coords) - 1) << 3)) 

156 out += [_zigzag(x - cx), _zigzag(y - cy)] 

157 cx, cy = x, y 

158 return out

161class Layer: 

162 """One layer of a vector tile: features with properties, sharing key and value tables.""" 

163 

164 def __init__(self, name: str, extent: int): 

165 """Start an empty layer. 

166 

167 Args: 

168 name: Layer name (the style's "source-layer"). 

169 extent: Tile coordinate range. 

170 """ 

171 self.name, self.extent = name, extent 

172 self.keys: dict[str, int] = {} 

173 self.values: dict[tuple, int] = {} 

174 self.features: list[bytes] = [] 

175 

176 def _tags(self, props: dict) -> list[int]: 

177 """Return the key/value index pairs of a feature's properties (None values are left out). 

178 

179 Args: 

180 props: Property name -> value. 

181 

182 Returns: 

183 [key index, value index, ...]. 

184 """ 

185 tags = [] 

186 for k, v in props.items(): 

187 if v is None: 

188 continue 

189 ki = self.keys.setdefault(k, len(self.keys)) 

190 vk = (type(v).__name__, v) 

191 vi = self.values.setdefault(vk, len(self.values)) 

192 tags += [ki, vi] 

193 return tags 

194 

195 def add(self, kind: str, coords: list, props: dict, fid: int | None = None) -> None: 

196 """Add a feature. 

197 

198 Args: 

199 kind: "point", "line" or "polygon". 

200 coords: Tile-local integer coordinates: one for a point, 2+ for a line; for a polygon its rings (outer 

201 first with positive ring_area, then its holes with negative), each 3+ vertices not repeating the 

202 first. 

203 props: Property name -> str / bool / int / float (None left out). 

204 fid: Optional feature id (>= 0). 

205 """ 

206 geom_type = {"point": _POINT, "line": _LINESTRING, "polygon": _POLYGON}[kind] 

207 msg = b"" 

208 if fid is not None: 

209 msg += _key(1, 0) + _varint(fid) 

210 msg += _packed(2, self._tags(props)) 

211 msg += _key(3, 0) + _varint(geom_type) 

212 geom = _polygon(coords) if geom_type == _POLYGON else _geometry(geom_type, coords) 

213 msg += _packed(4, geom) 

214 self.features.append(msg) 

215 

216 def encode(self) -> bytes: 

217 """Encode the layer message. 

218 

219 Returns: 

220 The Layer message bytes. 

221 """ 

222 msg = _key(15, 0) + _varint(2) + _bytes_field(1, self.name.encode("utf-8")) 

223 msg += b"".join(_bytes_field(2, f) for f in self.features) 

224 msg += b"".join(_bytes_field(3, k.encode("utf-8")) for k in self.keys) 

225 msg += b"".join(_bytes_field(4, _value(v)) for (_t, v) in self.values) 

226 msg += _key(5, 0) + _varint(self.extent) 

227 return msg
