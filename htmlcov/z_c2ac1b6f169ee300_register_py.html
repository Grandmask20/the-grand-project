17def neighbour_signature(pts: np.ndarray) -> np.ndarray: 

18 """Distances to the two nearest other map nodes: a cheap, zoom-stable local fingerprint. 

19 

20 Args: 

21 pts: (n, 2) map node positions. 

22 

23 Returns: 

24 (n, 2) array of the distances to the nearest and second-nearest other node (all 1e6 with fewer than 

25 3 points). 

26 """ 

27 if len(pts) < 3: 

28 return np.full((len(pts), 2), 1e6) 

29 from scipy.spatial import cKDTree 

30 

31 d, _ = cKDTree(pts).query(pts, k=3) # itself + the two nearest (no N x N matrix) 

32 return d[:, 1:3]

35def vote_translations(src: np.ndarray, dst: np.ndarray, top_k: int = 8): 

36 """Vote for the shifts that move a screenshot's map nodes onto the atlas's. 

37 

38 Every pair (screenshot node, atlas node) votes for the shift between them in a 2-D histogram of VOTE_BIN 

39 bins; pairs whose neighbour signatures agree (SIG_DIST1 / SIG_DIST2) count 1, the others 0.15. The 

40 histogram is smoothed over 3x3 bins and its highest peaks are taken, each blanking its surroundings. 

41 

42 Args: 

43 src: (n, 2) map nodes of the new screenshot (flat map coordinates). 

44 dst: (m, 2) map nodes of the atlas (atlas units). 

45 top_k: Most peaks to return. 

46 

47 Returns: 

48 List of (votes, tx, ty) peaks, strongest first; only peaks with more than 2 votes. 

49 """ 

50 s_sig, d_sig = neighbour_signature(src), neighbour_signature(dst) 

51 good = (np.abs(s_sig[:, None, 0] - d_sig[None, :, 0]) < SIG_DIST1) & ( 

52 np.abs(s_sig[:, None, 1] - d_sig[None, :, 1]) < SIG_DIST2 

53 ) 

54 weights = np.where(good, 1.0, 0.15).ravel() 

55 dx = (dst[None, :, 0] - src[:, None, 0]).ravel() 

56 dy = (dst[None, :, 1] - src[:, None, 1]).ravel() 

57 x0, x1, y0, y1 = dx.min(), dx.max(), dy.min(), dy.max() 

58 nx, ny = max(3, int((x1 - x0) / VOTE_BIN) + 1), max(3, int((y1 - y0) / VOTE_BIN) + 1) 

59 hist, xe, ye = np.histogram2d(dx, dy, bins=[nx, ny], range=[[x0, x1 + 1e-6], [y0, y1 + 1e-6]], weights=weights) 

60 smooth = cv2.boxFilter(hist.astype(np.float32), -1, (3, 3), normalize=False) 

61 peaks = [] 

62 work = smooth.copy() 

63 for _ in range(top_k): 

64 i, j = np.unravel_index(int(np.argmax(work)), work.shape) 

65 if work[i, j] <= 2: 

66 break 

67 peaks.append((float(work[i, j]), float(xe[i] + VOTE_BIN / 2), float(ye[j] + VOTE_BIN / 2))) 

68 work[max(0, i - 3) : i + 4, max(0, j - 3) : j + 4] = 0 

69 return peaks

72def fit_affine(src: np.ndarray, dst: np.ndarray) -> np.ndarray | None: 

73 """Least-squares placement; None when it would be an implausible camera change. 

74 

75 With the camera tilt undone a pan is a pure shift, so by default only a shift is fitted 

76 (PLACEMENT = "shift"). A free affine fitted ~1% squash and ~0.2 deg turns to the overlap, 

77 and those compounded from screenshot to screenshot into 30-45 unit offsets far from it. 

78 

79 Args: 

80 src: (n, 2) matched points of the screenshot (flat map coordinates). 

81 dst: (n, 2) the atlas points they match (atlas units). 

82 

83 Returns: 

84 2x3 affine src -> dst, or None when a free affine's scale falls outside SCALE_RANGE. 

85 """ 

86 if PLACEMENT == "shift" and K.HORIZON is not None: 

87 return np.hstack([np.eye(2), (dst - src).mean(axis=0)[:, None]]) 

88 A = np.hstack([src, np.ones((len(src), 1))]) 

89 sol, *_ = np.linalg.lstsq(A, dst, rcond=None) 

90 M = sol.T 

91 sv = np.linalg.svd(M[:, :2], compute_uv=False) 

92 if sv[1] < SCALE_RANGE[0] or sv[0] > SCALE_RANGE[1]: 

93 return None 

94 return M

97def mutual_pairs(a: np.ndarray, b: np.ndarray, tol: float): 

98 """Pair points of two sets that are each other's nearest point. 

99 

100 Args: 

101 a: (n, 2) points. 

102 b: (m, 2) points. 

103 tol: Largest distance of a pair (same units as the points). 

104 

105 Returns: 

106 List of (i, j, dist): a[i] and b[j] are each other's nearest point, within tol. 

107 """ 

108 if not len(a) or not len(b): 

109 return [] 

110 from scipy.spatial import cKDTree 

111 

112 da, ia = cKDTree(b).query(a) 

113 _db, ib = cKDTree(a).query(b) 

114 return [(i, int(ia[i]), float(da[i])) for i in range(len(a)) if ib[ia[i]] == i and da[i] <= tol]

117def refine(src: np.ndarray, dst: np.ndarray, t0) -> tuple[np.ndarray, list, float]: 

118 """Refine a shift into a placement by matching map nodes with shrinking tolerances. 

119 

120 Starting from the shift t0, each tolerance in REFINE_TOLS pairs the nodes (mutual_pairs) and refits: a 

121 fit_affine with 7+ pairs, otherwise the mean remaining shift. Stops early when fewer than 4 pairs remain. 

122 

123 Args: 

124 src: (n, 2) map nodes of the screenshot (flat map coordinates). 

125 dst: (m, 2) map nodes of the atlas (atlas units). 

126 t0: Starting shift (tx, ty) in atlas units. 

127 

128 Returns: 

129 (M, pairs, rmse): the 2x3 placement, the mutual pairs at the last tolerance as (i, j, dist), and their 

130 RMS distance in atlas units (99.0 when there are none). 

131 """ 

132 M = np.array([[1, 0, t0[0]], [0, 1, t0[1]]], float) 

133 pairs: list = [] 

134 for tol in REFINE_TOLS: 

135 pairs = mutual_pairs(apply(M, src), dst, tol) 

136 if len(pairs) < 4: 

137 break 

138 a = src[[p[0] for p in pairs]] 

139 b = dst[[p[1] for p in pairs]] 

140 fitted = fit_affine(a, b) if len(pairs) >= 7 else None 

141 if fitted is None: 

142 M = M.copy() 

143 M[:, 2] += (b - apply(M, a)).mean(axis=0) 

144 else: 

145 M = fitted 

146 pairs = mutual_pairs(apply(M, src), dst, REFINE_TOLS[-1]) 

147 rmse = float(np.sqrt(np.mean([p[2] ** 2 for p in pairs]))) if pairs else 99.0 

148 return M, pairs, rmse

152@dataclass 

153class Registration: 

154 """The result of trying to place a screenshot in the atlas. 

155 

156 Attributes: 

157 ok: True when the placement is accepted. 

158 reason: Why it was accepted or rejected (shown to the user). 

159 M: 2x3 affine: the screenshot's flat map coordinates -> atlas coords; None when nothing was found. 

160 inliers: Number of map nodes that agree (mutual pairs). 

161 rmse: RMS distance of those pairs, atlas units. 

162 overlap_src: Screenshot map nodes that land inside already placed screenshots. 

163 overlap_dst: Atlas map nodes that lie inside the new screenshot. 

164 ratio: inliers / the smaller overlap count. 

165 ncc: Best terrain correlation against an overlapping placed screenshot (see terrain_ncc). 

166 ncc_pixels: Number of pixels that correlation was measured on. 

167 ref_frame: Name of the screenshot the terrain was compared with. 

168 candidates: Up to 4 other tried placements as (inliers, rmse, ratio), for the log. 

169 """ 

170 

171 ok: bool 

172 reason: str 

173 M: np.ndarray | None = None 

174 inliers: int = 0 

175 rmse: float = 0.0 

176 overlap_src: int = 0 

177 overlap_dst: int = 0 

178 ratio: float = 0.0 

179 ncc: float = 0.0 

180 ncc_pixels: int = 0 

181 ref_frame: str = "" 

182 candidates: list = field(default_factory=list)

185def _inside_rect( 

186 pts: np.ndarray, M: np.ndarray, w: int, h: int, R: np.ndarray | None = None, margin: float = 30 

187) -> np.ndarray: 

188 """Tell which atlas points lie inside a placed screenshot. 

189 

190 Args: 

191 pts: (n, 2) atlas points. 

192 M: 2x3 placement of the screenshot (flat map coordinates -> atlas). 

193 w: Screenshot width in px. 

194 h: Screenshot height in px. 

195 R: The screenshot's 3x3 rectifier, or None when its coordinates are not flattened. 

196 margin: Points closer than this to the screenshot's edge (screen px) count as outside. 

197 

198 Returns: 

199 (n,) bool array: atlas points that lie inside the screenshot (0..w, 0..h) placed by M (after rectifier R). 

200 """ 

201 back = apply(invert(M), pts) 

202 if R is not None: 

203 back = persp(np.linalg.inv(R), back) 

204 return (back[:, 0] > margin) & (back[:, 0] < w - margin) & (back[:, 1] > margin) & (back[:, 1] < h - margin)

207def register( 

208 frame: Frame, atlas_pts: np.ndarray, footprints: list[tuple[np.ndarray, int, int, np.ndarray]] 

209) -> tuple[Registration, list]: 

210 """Find M (frame px -> atlas coords) from map nodes only. Terrain check is done by the caller. 

211 

212 Each shift peak from vote_translations is refined; the one with the most agreeing nodes (then the lowest 

213 RMSE) wins. 

214 

215 Args: 

216 frame: The new screenshot. 

217 atlas_pts: (m, 2) map nodes of the atlas (atlas units). 

218 footprints: Placed screenshots as (M, w, h, R), used to count the overlap. 

219 

220 Returns: 

221 (best, tried): the best Registration (ok is still False) and every tried one, best first 

222 (empty when there were too few nodes or no shift peak). 

223 """ 

224 src = frame.pts[frame.usable] 

225 if len(src) < MIN_INLIERS or len(atlas_pts) < MIN_INLIERS: 

226 return Registration(False, "too few nodes to match"), [] 

227 best, tried = None, [] 

228 for _votes, tx, ty in vote_translations(src, atlas_pts): 

229 M, pairs, rmse = refine(src, atlas_pts, (tx, ty)) 

230 inl = len(pairs) 

231 moved = apply(M, src) 

232 inside_src = np.zeros(len(src), bool) 

233 for FM, fw, fh, FR in footprints: 

234 inside_src |= _inside_rect(moved, FM, fw, fh, FR) 

235 inside_dst = _inside_rect(atlas_pts, M, frame.w, frame.h, frame.R) 

236 o_src, o_dst = int(inside_src.sum()), int(inside_dst.sum()) 

237 ratio = inl / max(1, min(o_src, o_dst)) 

238 cand = Registration(False, "", M, inl, rmse, o_src, o_dst, ratio) 

239 tried.append(cand) 

240 if best is None or (inl, -rmse) > (best.inliers, -best.rmse): 

241 best = cand 

242 if best is None: 

243 return Registration(False, "no translation peak"), [] 

244 tried.sort(key=lambda r: (-r.inliers, r.rmse)) 

245 return best, tried

248def _horizon_of(frame: Frame) -> float | None: 

249 """Recover the horizon a frame was rectified with (None = flat). 

250 

251 Args: 

252 frame: The screenshot. 

253 

254 Returns: 

255 The horizon row as a fraction of the screenshot height, or None when R is the identity. 

256 """ 

257 R = frame.R / frame.R[2, 1] if frame.R[2, 1] else frame.R 

258 if np.allclose(frame.R, np.eye(3)): 

259 return None 

260 return float(-R[2, 2] / frame.h)
