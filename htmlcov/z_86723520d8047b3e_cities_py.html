1"""City types (Ezomyte / Vaal / Faridun) and the maps around cities and Breach strongholds. 

2 

3City types 

4 Cities come in trios: one Ezomyte (grass, round citadel), one Vaal (big stepped pyramid) and one 

5 Faridun (desert, green ornate statues) a few maps apart. Two cities closer than TRIO_GAP atlas px 

6 were never of the same type on the test atlas (25 of 25 pairs). A trio is a close TRIANGLE (not a 

7 line); a complete trio is used up, and a city next to two known, still-free cities of different 

8 types that form a triangle with it gets the third type. Partners may be hidden in the fog, so this 

9 is a guess too. Older note: a city whose close neighbours 

10 are known gets the type that is left ("trio"). Otherwise it takes the type of the most alike 

11 labelled city in city_refs.npz (colour + edge-direction look of the ground around the city, 

12 "look-alike"). Faridun is easy to tell by its sand; Vaal vs Ezomyte by look alone is weak, so 

13 the trio rule and hand-set types (atlas.city_types, right-click on the map) win over it. 

14 

15Potential maps 

16 Maps within a few path steps of a City or a Breach stronghold may take on its biome. 

17"""

39def _edge_dirs(g: np.ndarray) -> np.ndarray: 

40 """Histogram the edge directions of a grey image in 16x16 px blocks. 

41 

42 Args: 

43 g: the grey image, 64x96 px (h x w). 

44 

45 Returns: 

46 A float array of 216 values: 9 direction bins (0-180 degrees, weighted by edge strength) for each of the 

47 4x6 blocks. 

48 """ 

49 g = g.astype(np.float32) 

50 gx, gy = cv2.Sobel(g, cv2.CV_32F, 1, 0), cv2.Sobel(g, cv2.CV_32F, 0, 1) 

51 mag = np.hypot(gx, gy) 

52 b = np.minimum(((np.arctan2(gy, gx) % np.pi) / np.pi * 9).astype(int), 8) 

53 return np.concatenate( 

54 [ 

55 np.bincount(b[i : i + 16, j : j + 16].ravel(), mag[i : i + 16, j : j + 16].ravel(), 9) 

56 for i in range(0, 64, 16) 

57 for j in range(0, 96, 16) 

58 ] 

59 )

62def window_feature(win: np.ndarray) -> np.ndarray: 

63 """Describe the look of the ground around a city base. 

64 

65 Args: 

66 win: BGR screenshot window ~540x360 px with the city's map base at the bottom middle. 

67 

68 Returns: 

69 A float32 feature vector: the square-rooted hue / saturation histogram (12x6, weighted by COLOR_W) 

70 followed by the normalised edge directions from _edge_dirs. 

71 """ 

72 win = cv2.resize(win, (192, 128), interpolation=cv2.INTER_AREA) 

73 hsv = cv2.cvtColor(win, cv2.COLOR_BGR2HSV) 

74 h = cv2.calcHist([hsv], [0, 1], None, [12, 6], [0, 180, 0, 256]).ravel() 

75 h /= h.sum() + 1e-9 

76 d = _edge_dirs(cv2.resize(cv2.cvtColor(win, cv2.COLOR_BGR2GRAY), (96, 64))) 

77 d /= np.linalg.norm(d) + 1e-9 

78 return np.concatenate([np.sqrt(h) * COLOR_W, d]).astype(np.float32)

81def city_features(atlas, x: float, y: float, n: int = 3) -> list[np.ndarray]: 

82 """Return the ground features of the city at atlas (x, y) from the n screenshots that show it most centrally. 

83 

84 Args: 

85 atlas: the core.Atlas with placed screenshots. 

86 x: x of the city in atlas units. 

87 y: y of the city in atlas units. 

88 n: how many screenshots to use at most. 

89 

90 Returns: 

91 A list of window_feature vectors, one per usable screenshot (missing files and windows cut off by the 

92 screenshot edge are skipped). 

93 """

122def look_distances(feats: list[np.ndarray]) -> list[float] | None: 

123 """Return the distance to the most alike labelled city of each type. 

124 

125 Args: 

126 feats: the city's features from city_features. 

127 

128 Returns: 

129 Three distances in TYPES order (9.0 for a type with no labelled city); None without a view or without 

130 labelled cities. 

131 """ 

132 R, T = _load_refs() 

133 if not feats or not len(R): 

134 return None 

135 best = [9.0, 9.0, 9.0] 

136 for u in feats: 

137 d = np.linalg.norm(R - u, axis=1) 

138 for k, t in enumerate(T): 

139 i = TYPES.index(t) 

140 best[i] = min(best[i], float(d[k])) 

141 return best

144def nearest_map(pts: np.ndarray, alive: np.ndarray, x: float, y: float) -> tuple[int, float]: 

145 """Find the nearest live map node to (x, y). 

146 

147 Args: 

148 pts: (N, 2) array of map node positions (atlas units). 

149 alive: bool array (N,), False for removed map nodes. 

150 x: x in atlas units. 

151 y: y in atlas units. 

152 

153 Returns: 

154 (index of the nearest live map node, its distance in atlas units); (-1, inf) with no maps. 

155 """ 

156 if not len(pts): 

157 return -1, float("inf") 

158 d = np.hypot(pts[:, 0] - x, pts[:, 1] - y) 

159 d[~alive] = np.inf 

160 v = int(np.argmin(d)) 

161 return v, float(d[v])

164def _near(items: list[dict], x, y, gap=SAME_CITY): 

165 """Find the stored item (hand-set type or cached look) closest to (x, y).

168 items: dicts with "x", "y" in atlas units. 

169 x: x in atlas units. 

170 y: y in atlas units. 

171 gap: the item must be closer than this (atlas units). 

172

174 The closest item dict, or None when none is closer than gap.

180 best, bd = it, d 

181 return best

184def city_types(atlas, cities: list[dict], compute_missing=True) -> list[tuple[str, str]]: 

185 """Decide the type (Ezomyte / Vaal / Faridun) of each city. 

186 

187 Hand-set types win; then the trio rule runs until nothing more follows; then the clearest look-alike 

188 (biggest margin between its two best allowed types) is decided next, and the trio rule runs again. 

189 Look-alike distances are cached in atlas.city_auto (computed for new cities when allowed). 

190 

191 Args: 

192 atlas: the core.Atlas (uses atlas.city_types and atlas.city_auto; city_auto may grow). 

193 cities: city dicts with "x", "y" in atlas units. 

194 compute_missing: True to compute look-alike distances for cities without a cached one. 

195 

196 Returns: 

197 One (type, how) per city; how is "set" (by hand), "trio" or "look-alike"; ("", "") when undecided. 

198 """ 

199 n = len(cities) 

200 out: list[tuple[str, str] | None] = [None] * n 

201 dist: list[list[float] | None] = [None] * n 

202 for k, c in enumerate(cities): 

203 s = _near(atlas.city_types, c["x"], c["y"]) 

204 if s: 

205 out[k] = (s["type"], "set") 

206 continue 

207 a = _near(atlas.city_auto, c["x"], c["y"]) 

208 if a is None and compute_missing: 

209 d = look_distances(city_features(atlas, c["x"], c["y"])) 

210 if d is not None: 

211 a = {"x": round(c["x"], 1), "y": round(c["y"], 1), "d": [round(v, 4) for v in d]} 

212 atlas.city_auto.append(a) 

213 dist[k] = a["d"] if a else None 

214 P = np.array([[c["x"], c["y"]] for c in cities], float).reshape(-1, 2) 

215 D = np.hypot(P[:, None, 0] - P[None, :, 0], P[:, None, 1] - P[None, :, 1]) if n else np.zeros((0, 0)) 

216 

217 def allowed(k): 

218 """List the types city k may still have (closer cities, under TRIO_GAP, always differ in type). 

219 

220 Args: 

221 k: index of the city in cities. 

222 

223 Returns: 

224 The TYPES not taken by a decided city closer than TRIO_GAP. 

225 """ 

226 taken = {out[j][0] for j in np.nonzero(D[k] < TRIO_GAP)[0] if j != k and out[j]} 

227 return [t for t in TYPES if t not in taken]

229 def triangle(i, j, k): 

230 """Check that three cities form a trio: a close TRIANGLE (all sides < TRIO_SIDE, no angle under MIN_ANGLE). 

231 

232 Args: 

233 i: index of the first city in cities. 

234 j: index of the second city. 

235 k: index of the third city. 

236 

237 Returns: 

238 The triangle's perimeter in atlas units, or None when it is not a trio (too big or nearly a line). 

239 """ 

240 a_, b_, c_ = D[j, k], D[i, k], D[i, j] 

241 if max(a_, b_, c_) >= TRIO_SIDE: 

242 return None 

243 cosmax = max( 

244 (b_ * b_ + c_ * c_ - a_ * a_) / (2 * b_ * c_ + 1e-9), 

245 (a_ * a_ + c_ * c_ - b_ * b_) / (2 * a_ * c_ + 1e-9), 

246 (a_ * a_ + b_ * b_ - c_ * c_) / (2 * a_ * b_ + 1e-9), 

247 ) 

248 if cosmax > np.cos(np.radians(MIN_ANGLE)): 

249 return None 

250 return a_ + b_ + c_ 

251 

252 in_trio: set[int] = set() 

253 tris = [] 

254 for i in range(n): 

255 nb = [j for j in np.nonzero(D[i] < TRIO_SIDE)[0] if j > i] 

256 for x_, j in enumerate(nb): 

257 for k in nb[x_ + 1 :]: 

258 if triangle(i, j, k) is not None: 

259 tris.append((i, j, k)) 

260 

261 def close_trios(): 

262 """Mark complete trios (three types, one triangle) as used up: their cities decide nothing else.""" 

263 for i, j, k in tris: 

264 if out[i] and out[j] and out[k] and len({out[i][0], out[j][0], out[k][0]}) == 3: 

265 in_trio.update((i, j, k)) 

266 

267 while True: 

268 changed = True 

269 while changed: # trio rule until nothing more follows from it 

270 changed = False 

271 close_trios() 

272 for k in range(n): 

273 if out[k] is not None: 

274 continue 

275 best_t = None # the tightest triangle with two known, free cities 

276 near = [j for j in np.nonzero(D[k] < TRIO_SIDE)[0] if j != k and out[j] and j not in in_trio] 

277 for x_, j1 in enumerate(near): 

278 for j2 in near[x_ + 1 :]: 

279 if out[j1][0] == out[j2][0]: 

280 continue 

281 per = triangle(k, j1, j2) 

282 if per is not None and (best_t is None or per < best_t[0]): 

283 best_t = (per, j1, j2) 

284 if best_t: 

285 left = [t for t in TYPES if t not in (out[best_t[1]][0], out[best_t[2]][0])] 

286 out[k] = (left[0], "trio") 

287 in_trio.update((k, best_t[1], best_t[2])) 

288 changed = True 

289 open_ = [k for k in range(n) if out[k] is None and dist[k] is not None] 

290 if not open_: 

291 break 

292 # decide the clearest look-alike next (biggest margin between the two best allowed types) 

293 best = None 

294 for k in open_: 

295 al = allowed(k) or list(TYPES) 

296 ds = sorted((dist[k][TYPES.index(t)], t) for t in al) 

297 margin = ds[1][0] - ds[0][0] if len(ds) > 1 else 9.0 

298 if best is None or margin > best[0]: 

299 best = (margin, k, ds[0][1]) 

300 out[best[1]] = (best[2], "look-alike") 

301 return [o if o else ("", "") for o in out]

321def ground_sampler(canvas: np.ndarray, weight: np.ndarray, origin, radius: float = 70.0): 

322 """Make a function that tells the look of the ground around an atlas point, from the stitched map image. 

323 

324 Args: 

325 canvas: the stitched map image (BGR). 

326 weight: per-pixel coverage of the canvas, same height / width; 0 where nothing was seen. 

327 origin: (x0, y0, px per unit): the atlas point at the canvas's top-left and the canvas scale. 

328 radius: radius of the sampled disc in atlas units. 

329 

330 Returns: 

331 A function (x, y) -> 6 floats (Lab mean + spread) for an atlas point; it returns None where the disc 

332 leaves the canvas or less than half of it was seen. 

333 """ 

334 x0, y0, s0 = origin 

335 lab = cv2.cvtColor(canvas, cv2.COLOR_BGR2LAB).astype(np.float32) 

336 r = max(2, int(round(radius * s0))) 

337 yy, xx = np.mgrid[-r : r + 1, -r : r + 1] 

338 disc = np.hypot(xx, yy) <= r 

339 H, W = canvas.shape[:2] 

340 

341 def at(x, y): 

342 """Return the ground look (Lab mean + spread) around atlas point (x, y). 

343 

344 Args: 

345 x: x in atlas units. 

346 y: y in atlas units. 

347 

348 Returns: 

349 A 6-float array, or None where the disc leaves the canvas or less than half of it was seen. 

350 """ 

351 cx, cy = int(round((x - x0) * s0)), int(round((y - y0) * s0)) 

352 if cx - r < 0 or cy - r < 0 or cx + r + 1 > W or cy + r + 1 > H: 

353 return None 

354 m = disc & (weight[cy - r : cy + r + 1, cx - r : cx + r + 1] > 0) 

355 if m.sum() < disc.sum() * 0.5: 

356 return None 

357 px = lab[cy - r : cy + r + 1, cx - r : cx + r + 1][m] 

358 return np.concatenate([px.mean(0), px.std(0)]) # spread: Faridun ground is flat, desert mottled 

359 

360 dy, dx = np.mgrid[-r : r + 1, -r : r + 1] 

361 need = disc.sum() * 0.5 

362 

363 def many(points) -> tuple[np.ndarray, np.ndarray]: 

364 """`at` for many atlas points at once (one call per map node took ~2 s on a 24,000-map atlas, user). 

365 

366 Args: 

367 points: (n, 2) atlas x, y. 

368 

369 Returns: 

370 (looks, ok): (n, 6) Lab mean + spread, and (n,) whether each point has one (where `at` gives None, the 

371 row is zeros and ok is False). 

372 """ 

373 P = np.asarray(points, float).reshape(-1, 2) 

374 out = np.zeros((len(P), 6), np.float32) 

375 ok = np.zeros(len(P), bool) 

376 cx = np.round((P[:, 0] - x0) * s0).astype(int) 

377 cy = np.round((P[:, 1] - y0) * s0).astype(int) 

378 inside = np.nonzero((cx - r >= 0) & (cy - r >= 0) & (cx + r + 1 <= W) & (cy + r + 1 <= H))[0] 

379 # as many discs as fit the pixel budget (a fixed 512 was 300 MB with big discs: CodeRabbit), at least one 

380 chunk = max(1, CITY_GROUND_PIXELS // (2 * r + 1) ** 2) 

381 for k in range(0, len(inside), chunk): 

382 i = inside[k : k + chunk] 

383 ys, xs = cy[i, None, None] + dy, cx[i, None, None] + dx # (m, 2r+1, 2r+1) each 

384 m = (weight[ys, xs] > 0) & disc 

385 cnt = m.sum((1, 2)) 

386 good = cnt >= need 

387 if not good.any(): 

388 continue 

389 i, ys, xs, m, cnt = i[good], ys[good], xs[good], m[good], cnt[good] 

390 px = lab[ys, xs] # (m, 2r+1, 2r+1, 3) float32 

391 w = m.astype(np.float32) 

392 # one pass: sums and sums of squares over the seen part of each disc (totals in float64) 

393 s1 = np.einsum("nyxc,nyx->nc", px, w, dtype=np.float64) 

394 s2 = np.einsum("nyxc,nyxc,nyx->nc", px, px, w, dtype=np.float64) 

395 mean = s1 / cnt[:, None] 

396 std = np.sqrt(np.maximum(s2 / cnt[:, None] - mean**2, 0.0)) 

397 out[i, :3], out[i, 3:], ok[i] = mean, std, True 

398 return out, ok 

399 

400 at.many = many # type: ignore[attr-defined] 

401 return at

433def square_buttons(img) -> list[dict]: 

434 """Find the square framed buttons on the atlas in one screenshot. 

435 

436 The locked stronghold's hand icon and the Expedition ship marker share the frame (user). Buttons are found 

437 by their frame (>= BUTTON_FRAME, not inside the HUD), then named by the symbol inside. Full size, ~70 ms. 

438 

439 Args: 

440 img: the screenshot (BGR or grey). 

441 

442 Returns: 

443 A list of dicts with "kind" ('lock', 'ship', or 'unknown' for a symbol not taught yet), "x", "y" (the 

444 button's centre in screen px), "frame" and "symbol" (match scores). 

445 """ 

446 R = _button_refs() 

447 t = R["frame"] 

448 h, w = t.shape 

449 g = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY) if img.ndim == 3 else img 

450 r = np.nan_to_num(cv2.matchTemplate(g, t, cv2.TM_CCORR_NORMED, mask=R["mask"]), nan=0, posinf=0, neginf=0) 

451 # HUD art: only matches whose centre is well inside the HUD outline (a button can sit right next to it) 

452 hud = cv2.erode((core.hud_mask(g.shape[1], g.shape[0]) > 0).astype(np.uint8), np.ones((41, 41), np.uint8)) > 0 

453 r[hud[h // 2 : h // 2 + r.shape[0], w // 2 : w // 2 + r.shape[1]]] = 0 

454 out = [] 

455 while True: 

456 _mn, mx, _l, (x, y) = cv2.minMaxLoc(r) 

457 if mx < BUTTON_FRAME: 

458 return out 

459 r[max(0, y - h) : y + h, max(0, x - w) : x + w] = 0 

460 c = g[y + 14 - BUTTON_EDGE : y + h - 14 + BUTTON_EDGE, x + 14 - BUTTON_EDGE : x + w - 14 + BUTTON_EDGE] 

461 best, kind = -1.0, "unknown" 

462 for k, sy in R["symbols"].items(): 

463 if c.shape[0] < sy.shape[0] or c.shape[1] < sy.shape[1]: 

464 continue 

465 v = float(cv2.matchTemplate(c, sy, cv2.TM_CCOEFF_NORMED).max()) 

466 if v > best: 

467 best, kind = v, k 

468 if best < BUTTON_SYMBOL: 

469 if mx < 0.9: # a weak frame with no known symbol: not a button 

470 continue 

471 kind = "unknown" 

472 out.append({"kind": kind, "x": x + w / 2, "y": y + h / 2, "frame": round(mx, 3), "symbol": round(best, 3)})

475def locks_in(img) -> list[tuple[float, float]]:

535def potential_maps( 

536 nodes: list[dict], 

537 paths: list[dict], 

538 sources: list[tuple[float, float, str]], 

539 count: int = 10, 

540 hops: int = 3, 

541 snap: float = 90.0, 

542 ground=None, 

543 area_kinds: tuple = ("Breach stronghold",), 

544 area_reach: float = 520.0, 

545 area_blind: float = 230.0, 

546 area_merge: float = 350.0, 

547 area_hops: int = 6, 

548 area_snap: float = 260.0, 

549 pool_r: float = 260.0, 

550 far_r: float = 700.0, 

551 alike: float = 10.0, 

552 rich: float = 5.0, 

553 poor: float = 1.5, 

554 blocked: dict[int, str] | None = None, 

555 unique_ok: tuple = ("Vaal city",), 

556 hard_off: dict | None = None, 

557 behind: float = -0.2, 

558) -> dict[int, tuple[str, int]]: 

559 """Find the maps a City / Breach stronghold biome may cover. 

560 

561 Cities: about 10 maps (+-2), linked tightly to each other (often a square grid of paths) with 

562 few paths leading out. Each city grows such a cluster from its own map (step 0: the nearest map 

563 within `snap` px), at most `hops` steps out, preferring maps with more links into the cluster, 

564 maps that close a square (4-map loop) with it, and maps on the city's kind of ground; the size 

565 in count+-2 that leaves the fewest paths out (and the least foreign ground) is kept. 

566 

567 Breach strongholds (`area_kinds`): no size limit; a roughly round patch of their own ground: 

568 stronghold boxes within `area_merge` are one stronghold (its centre = their mean), and every map 

569 within `area_reach` of the centre reached through such maps (`area_blind` when there are too 

570 few strongholds to learn the Breach ground from). 

571 

572 Ground (see ground_sampler): each kind of city has its own ground everywhere (flat bright sand 

573 for Faridun, dark purple rock for Breach, ...), but the city's own map sits on dark buildings 

574 and often at the edge of its patch, so the ground is LEARNED per label: how much more common a 

575 map's ground look (within `alike` Lab units) is among maps within `pool_r` of all sources with 

576 that label than among maps farther than `far_r` from every source. At least `rich` times more 

577 common = this kind's ground; `poor` or less = clearly not (crawl atlas, Faridun: sand maps 4-17, 

578 maps across the border 1-2). Labels with fewer than 3 sources get no ground term. 

579 Rules from the user (cities): a city never connects to a Unique map, Tower, merchant, 

580 corruption / cleansed map, Hall, Breach stronghold, hideout or Abyss map, and nothing reached 

581 only through one of those is a city map (`blocked`: node -> reason; the exception is a Unique 

582 map inside a Vaal city, `unique_ok`, taken but not grown through). A city's citadel (its own 

583 map) is always at an EDGE of its biome, so maps behind it (direction against the side where 

584 its own kind of ground and maps lie, cosine below `behind`) are left out. Maps on clearly 

585 foreign ground are left out for Faridun (always desert; `hard_off`), elsewhere only penalised. 

586 A map claimed by two sources keeps the nearer; ties list both ("A / B"). 

587 

588 Args: 

589 nodes: map node dicts with "x", "y" (atlas units) and "status" ("removed" ones are skipped). 

590 paths: path dicts with "a", "b" (indices into nodes). 

591 sources: (x, y, label) of each City / Breach stronghold in atlas units, label e.g. "Faridun city". 

592 count: a city's usual number of maps (the kept size is within count +-2). 

593 hops: how many path steps a city reaches out at most. 

594 snap: a city's own map must be this close to it (atlas units). 

595 ground: a ground_sampler function, or None for no ground term. 

596 area_kinds: labels grown as a round patch with no size limit (Breach strongholds). 

597 area_reach: patch radius around a stronghold's centre (atlas units). 

598 area_blind: patch radius when the Breach ground cannot be learned (atlas units). 

599 area_merge: stronghold boxes closer than this are one stronghold (atlas units). 

600 area_hops: how many path steps a stronghold patch reaches out at most. 

601 area_snap: a stronghold's own map must be this close to it (atlas units). 

602 pool_r: maps within this of a source make up its label's ground sample (atlas units). 

603 far_r: maps farther than this from every source make up the background sample (atlas units). 

604 alike: two ground looks closer than this are alike (Lab units). 

605 rich: at least this many times more common = this kind's ground. 

606 poor: this many times or less = clearly not this kind's ground. 

607 blocked: map node index -> reason it is never a city map (see blocked_maps), or None. 

608 unique_ok: city labels whose biome may hold a Unique map (taken but not grown through). 

609 hard_off: label -> ground mismatch (0-2) from which maps are cut; None uses {"Faridun city": 1.6}. 

610 behind: cosine below which a map lies behind the citadel and is left out. 

611 

612 Returns: 

613 Map node index -> (label, path steps from the source's own map). 

614 """ 

615 pts = np.array([[nd["x"], nd["y"]] for nd in nodes], float).reshape(-1, 2) 

616 alive = np.array([nd.get("status") != "removed" for nd in nodes], bool) 

617 adj: dict[int, set[int]] = {} 

618 for p in paths: 

619 a, b = p["a"], p["b"] 

620 if alive[a] and alive[b] and a != b: 

621 adj.setdefault(a, set()).add(b) 

622 adj.setdefault(b, set()).add(a) 

623 res: dict[int, tuple[str, int]] = {} 

624 if not len(pts): 

625 return res 

626 gcache: dict[int, np.ndarray | None] = {} 

627 

628 def g(v): 

629 """Return the ground look of map node v (cached), or None without a sampler or where nothing was seen. 

630 

631 Args: 

632 v: index of the map node in nodes. 

633 

634 Returns: 

635 A 6-float array from the ground sampler, or None. 

636 """ 

637 if ground is None: 

638 return None 

639 if v not in gcache: 

640 gcache[v] = ground(*pts[v]) 

641 return gcache[v] 

642 

643 merged, used = [], set() # stronghold boxes close together = one stronghold 

644 for k, (x, y, label) in enumerate(sources): 

645 if k in used: 

646 continue 

647 if label in area_kinds: 

648 grp = [ 

649 j 

650 for j, (x2, y2, l2) in enumerate(sources) 

651 if j not in used and l2 == label and np.hypot(x2 - x, y2 - y) < area_merge 

652 ] 

653 used.update(grp) 

654 merged.append( 

655 (float(np.mean([sources[j][0] for j in grp])), float(np.mean([sources[j][1] for j in grp])), label) 

656 ) 

657 else: 

658 merged.append((x, y, label)) 

659 sources = merged 

660 starts = [] 

661 for x, y, label in sources: 

662 s, ds = nearest_map(pts, alive, x, y) 

663 if ds <= (area_snap if label in area_kinds else snap): 

664 starts.append((s, x, y, label)) 

665 enrich: dict[str, tuple[np.ndarray, np.ndarray]] = {} 

666 if ground is not None and starts: 

667 GV = np.zeros((len(pts), 6), np.float32) 

668 ok = np.zeros(len(pts), bool) 

669 live = np.nonzero(alive)[0] 

670 if hasattr(ground, "many"): # every map node's ground at once (one by one took ~2 s on rc1, user) 

671 looks, seen = ground.many(np.asarray(pts, float)[live]) 

672 GV[live[seen]], ok[live[seen]] = looks[seen], True 

673 for j, v in enumerate(live): 

674 gcache.setdefault(int(v), looks[j] if seen[j] else None) 

675 else: 

676 for v in live: 

677 gv = g(v) 

678 if gv is not None: 

679 GV[v], ok[v] = gv, True 

680 S = np.array([[x, y] for _s, x, y, _l in starts], float) 

681 dmin = np.full(len(pts), 1e9) 

682 for k in range(0, len(S), 64): 

683 dmin = np.minimum( 

684 dmin, 

685 np.hypot(pts[:, None, 0] - S[None, k : k + 64, 0], pts[:, None, 1] - S[None, k : k + 64, 1]).min( 

686 axis=1 

687 ), 

688 ) 

689 B = GV[ok & (dmin > far_r)] 

690 for label in {lab for *_r, lab in starts}: 

691 T = np.array([[x, y] for _s, x, y, lab in starts if lab == label], float) 

692 if len(T) < 3 or len(B) < 50: 

693 continue 

694 dl = np.full(len(pts), 1e9) 

695 for k in range(0, len(T), 64): 

696 dl = np.minimum( 

697 dl, 

698 np.hypot(pts[:, None, 0] - T[None, k : k + 64, 0], pts[:, None, 1] - T[None, k : k + 64, 1]).min( 

699 axis=1 

700 ), 

701 ) 

702 enrich[label] = (GV[ok & (dl < pool_r)], B) 

703 rcache: dict[tuple[str, int], float] = {}

705 def ratio(label, v): 

706 """How many times more common map node v's ground look is near this label's sources than far from all. 

707 

708 Args: 

709 label: the source label (must have a learned ground sample in enrich). 

710 v: index of the map node in nodes. 

711 

712 Returns: 

713 The ratio as a float (cached). 

714 """ 

715 key = (label, v) 

716 if key not in rcache: 

717 Pt, B = enrich[label] 

718 x = g(v) 

719 a = (np.linalg.norm(Pt - x, axis=1) < alike).mean() 

720 b = (np.linalg.norm(B - x, axis=1) < alike).mean() 

721 rcache[key] = float((a + 1e-3) / (b + 1e-3)) 

722 return rcache[key] 

723 

724 for s, x, y, label in starts: 

725 

726 def off(v, label=label): # 0 = this kind's ground ... 2 = clearly not 

727 """Score how foreign map node v's ground is for this label. 

728 

729 Args: 

730 v: index of the map node in nodes. 

731 label: the source label. 

732 

733 Returns: 

734 0.0 = this kind's ground (or no ground term) ... 2.0 = clearly not. 

735 """ 

736 if label not in enrich or g(v) is None: 

737 return 0.0 

738 r = ratio(label, v) 

739 return float(np.clip(np.log(rich / r) / np.log(rich / poor), 0.0, 1.0)) * 2.0 

740 

741 seen = {s: 0} # path steps from the source map 

742 front = [s] 

743 if label in area_kinds: 

744 # a round patch around the stronghold: distance from its centre, tighter when its 

745 # ground can't be checked (fewer than 3 strongholds to learn the Breach ground from) 

746 reach = area_reach if label in enrich else area_blind 

747 for step in range(1, area_hops + 1): 

748 nxt = [] 

749 for u in front: 

750 for v in adj.get(u, ()): 

751 if ( 

752 v not in seen 

753 and off(v) < 1.0 

754 and (blocked or {}).get(v) not in ("Hall", "Tower", "another city") 

755 and np.hypot(pts[v, 0] - x, pts[v, 1] - y) <= reach 

756 ): 

757 seen[v] = step 

758 nxt.append(v) 

759 front = nxt 

760 keep = list(seen) 

761 else: 

762 blk = blocked or {} 

763 

764 def barred(v, blocked_maps=blk, city_start=s, city_label=label): # never a city map 

765 """Tell whether map node v can never be a map of this city. 

766 

767 Args: 

768 v: index of the map node in nodes. 

769 blocked_maps: map node index -> reason it is never a city map. 

770 city_start: index of the city's own map (never barred). 

771 city_label: the city's label (a Unique map is allowed in unique_ok labels). 

772 

773 Returns: 

774 True when v is barred. 

775 """ 

776 r = blocked_maps.get(v) 

777 return v != city_start and r is not None and not (r == "Unique map" and city_label in unique_ok) 

778 

779 def dead_end(v, blocked_maps=blk, city_start=s): # a Unique map in a Vaal city: taken, not grown through 

780 """Tell whether the city's growth stops at map node v (a Unique map). 

781 

782 Args: 

783 v: index of the map node in nodes. 

784 blocked_maps: map node index -> reason it is never a city map. 

785 city_start: index of the city's own map (never a dead end). 

786 

787 Returns: 

788 True for a Unique map other than the city's own. 

789 """ 

790 return v != city_start and blocked_maps.get(v) == "Unique map" 

791 

792 # hard ground cut only where the user says the ground is always the same (Faridun = 

793 # desert); Ezomyte / Vaal ground is too varied, it only counts as a penalty there 

794 cut_at = (hard_off if hard_off is not None else {"Faridun city": 1.6}).get(label, 9.0) 

795 for step in range(1, hops + 1): 

796 nxt = [] 

797 for u in front: 

798 if dead_end(u): 

799 continue 

800 for v in adj.get(u, ()): 

801 if v not in seen and not barred(v) and off(v) < cut_at: 

802 seen[v] = step 

803 nxt.append(v) 

804 front = nxt 

805 # which side of the citadel the city lies on: where its maps and its ground are 

806 vec = np.zeros(2) 

807 for v, st in seen.items(): 

808 d = pts[v] - pts[s] 

809 n = float(np.hypot(*d)) 

810 if v != s and n > 1: 

811 vec += (2.0 - off(v)) / (1 + st) * d / n 

812 if np.hypot(*vec) > 0.8: 

813 u = vec / np.hypot(*vec) 

814 for v in [v for v in seen if v != s]: 

815 d = pts[v] - pts[s] 

816 if float(d @ u) / max(1.0, float(np.hypot(*d))) < behind: 

817 del seen[v] 

818 grown, best = [s], None 

819 inside = {s} 

820 mismatch = 0.0 

821 

822 def squares(c, cluster=inside): # 4-map loops c would close with the cluster 

823 """Count the 4-map loops (squares) map node c would close with the cluster. 

824 

825 Args: 

826 c: index of the candidate map node in nodes. 

827 cluster: the set of map node indices already in the city. 

828 

829 Returns: 

830 The number of squares. 

831 """ 

832 nb = [w for w in adj[c] if w in cluster] 

833 return sum(len((adj[a] & adj[b] & cluster) - {c}) for i, a in enumerate(nb) for b in nb[i + 1 :]) 

834 

835 while len(grown) < count + 3: 

836 cand = {v for u in inside if not dead_end(u) for v in adj.get(u, ()) if v not in inside and v in seen} 

837 if not cand: 

838 break 

839 v = max( 

840 cand, 

841 key=lambda c: ( 

842 len(adj[c] & inside) + 0.5 * min(2, squares(c)) - 1.5 * off(c), 

843 -seen[c], 

844 -float(np.hypot(*(pts[c] - pts[s]))), 

845 ), 

846 ) 

847 grown.append(v) 

848 inside.add(v) 

849 mismatch += off(v) 

850 if len(grown) - 1 >= count - 2: 

851 cut = sum(len(adj.get(u, set()) - inside) for u in inside) 

852 cost = cut + 1.5 * mismatch 

853 if best is None or cost < best[0]: 

854 best = (cost, len(grown)) 

855 keep = grown[: best[1]] if best else grown 

856 for v in keep: 

857 st = seen[v] 

858 old = res.get(v) 

859 if old is None or st < old[1]: 

860 res[v] = (label, st) 

861 elif st == old[1] and label not in old[0].split(" / "): 

862 res[v] = (f"{old[0]} / {label}", st) 

863 return res

884def icon_maps(nodes: list[dict], icons: list[dict], reach: float = 45.0) -> dict[int, int]: 

885 """Find the map node each icon belongs to (nearest live map node to the icon's base, within `reach`). 

886 

887 Args: 

888 nodes: map node dicts with "x", "y" (atlas units) and "status". 

889 icons: icon dicts with "x", "y" and optionally "bx", "by" (the base under the icon), atlas units. 

890 reach: the farthest a map node may be from the icon's base (atlas units). 

891 

892 Returns: 

893 Icon index -> map node index, for icons with a map node in reach. 

894 """ 

895 pts = np.array([[n["x"], n["y"]] for n in nodes], float).reshape(-1, 2) 

896 alive = np.array([n.get("status") != "removed" for n in nodes], bool) 

897 out = {} 

898 if not len(pts) or not alive.any() or not icons: 

899 return out 

900 from scipy.spatial import cKDTree # (v1.03: one map search per badge took ~1 s per redraw) 

901 

902 live = np.nonzero(alive)[0] 

903 q = np.array([[ic.get("bx", ic["x"]), ic.get("by", ic["y"])] for ic in icons], float) 

904 d, k = cKDTree(pts[live]).query(q) 

905 for i in np.nonzero(d <= reach)[0]: 

906 out[int(i)] = int(live[k[i]]) 

907 return out

910def blocked_maps( 

911 nodes: list[dict], 

912 icons: list[dict], 

913 landmarks: list[dict], 

914 icon_node: dict[int, int] | None = None, 

915 snap: float = 90.0, 

916) -> dict[int, str]: 

917 """Find the maps that are never city maps (user's rule), from their icons, flags and nearby landmarks. 

918 

919 Args: 

920 nodes: map node dicts with "x", "y" (atlas units), "status", and optionally "icons" (raw icon names), 

921 "tower" and "abyss". 

922 icons: icon dicts with "name" and positions (atlas units). 

923 landmarks: landmark dicts with "kind", "x", "y" and "status". 

924 icon_node: icon index -> map node index (from icon_maps), or None to compute it. 

925 snap: a landmark's map node must be this close to it (atlas units). 

926 

927 Returns: 

928 Map node index -> reason (Unique map, merchant, corruption, cleansed, hideout, Abyss, Tower, Hall, 

929 Breach stronghold, another city). 

930 """ 

931 out: dict[int, str] = {} 

932 for v, n in enumerate(nodes): 

933 if n.get("status") == "removed": 

934 continue 

935 for raw in n.get("icons", []): 

936 if raw in BLOCK_ICONS: 

937 out.setdefault(v, BLOCK_ICONS[raw]) 

938 if n.get("tower"): 

939 out.setdefault(v, "Tower") 

940 if n.get("abyss"): 

941 out.setdefault(v, "Abyss") 

942 for i, v in (icon_node if icon_node is not None else icon_maps(nodes, icons)).items(): 

943 if icons[i]["name"] in BLOCK_ICONS: 

944 out.setdefault(v, BLOCK_ICONS[icons[i]["name"]]) 

945 pts = np.array([[n["x"], n["y"]] for n in nodes], float).reshape(-1, 2) 

946 alive = np.array([n.get("status") != "removed" for n in nodes], bool) 

947 for lm in landmarks: 

948 if lm["kind"] not in BLOCK_PLACES or lm.get("status", "").startswith("removed") or not len(pts): 

949 continue 

950 v, dv = nearest_map(pts, alive, lm["x"], lm["y"]) 

951 if dv <= snap: 

952 out.setdefault(v, BLOCK_PLACES[lm["kind"]]) 

953 return out

956def near_city( 

957 nodes: list[dict], paths: list[dict], potential: dict[int, tuple[str, int]], blocked: dict[int, str] 

958) -> dict[int, tuple[str, int]]: 

959 """Return the potential city maps: those on a city's biome and those one path step from it. 

960 

961 Args: 

962 nodes: map node dicts with "status" ("removed" ones are skipped). 

963 paths: path dicts with "a", "b" (indices into nodes). 

964 potential: the result of potential_maps (only labels containing "city" are used). 

965 blocked: map node index -> reason it is never a city map; those are not added as neighbours. 

966 

967 Returns: 

968 Map node index -> (city label, 0 = on its biome / 1 = one path step from it). 

969 """ 

970 alive = [n.get("status") != "removed" for n in nodes] 

971 adj: dict[int, set[int]] = {} 

972 for p in paths: 

973 a, b = p["a"], p["b"] 

974 if alive[a] and alive[b]: 

975 adj.setdefault(a, set()).add(b) 

976 adj.setdefault(b, set()).add(a) 

977 out = {v: (lab, 0) for v, (lab, _st) in potential.items() if "city" in lab.lower()} 

978 for v, (lab, _z) in list(out.items()): 

979 for w in adj.get(v, ()): 

980 if w not in out and w not in blocked: 

981 out[w] = (lab, 1) 

982 return out

985def biomes(atlas, V: dict, ground=None, kept: bool = True) -> dict: 

986 """Work out the city / stronghold biomes of an atlas (the app's model and map use it). 

987 

988 In order (as the old window did: add_locks, rebuild_cities, apply_biome_edits): locked Breach 

989 strongholds from the hand icons, city types, the maps each City or unlocked stronghold may cover, then the 

990 biomes drawn by hand on top. 

991 

992 Args: 

993 atlas: The atlas.
