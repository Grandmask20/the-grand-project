1"""Clean terrain ("Fix seams"): the stitched map built with care instead of fast. 

2 

3The quick map (Atlas.composite / paste_into) takes every spot from the screenshot that shows it most centrally. 

4That cuts map nodes and their icons in half where two screenshots meet, lets a screenshot taken in fog paint 

5dark blocks over lit ground, and puts seams across paths and tall terrain. Here an ownership map ("labels": 

6which screenshot each spot comes from) is worked out first, on a grid of SEAM_GRID atlas units, by trading off: 

7 

8- each spot wants a screenshot that shows it centrally and lit (fogged ground scores low, so a lit copy wins) 

9- each seam costs, per cell of its length, more 

10 * the farther the spot is from the middle of the two screenshots that meet there: the game's view leans 

11 tall things (mountains, towers) outwards from the middle of the screen, so two views of a spot far from 

12 their middles don't line up, worst when they come from opposing edges 

13 * the farther apart the two screenshots were taken (the same thing, from the views' side) 

14 * through busy ground (mountains, trees, buildings: where the lean shows), much more through paths and 

15 most through map nodes and their icons (so those come whole from one screenshot) 

16 

17The cheapest ownership map overall is found with graph cuts (alpha expansion: for each screenshot in turn, 

18the best set of spots to hand over to it), then the map is drawn from it. The labels are kept with the map 

19(same grid as the map), so screenshots added quickly afterwards only take over where they are more central 

20than what is there (paste_into), and the zoomed-in view follows the same ownership. 

21"""

62def _score(img: np.ndarray) -> tuple[np.ndarray, np.ndarray]: 

63 """Score a screenshot for the clean map, at 1/SEAM_SCORE_SCALE size. 

64 

65 Args: 

66 img: BGR screenshot. 

67 

68 Returns: 

69 (score, busyness): float32 arrays at 1/SEAM_SCORE_SCALE size. score = centrality (distance from the 

70 interface / screenshot edge, in full-size px) x how lit (SEAM_DARK..SEAM_LIT, floor SEAM_DARK_FLOOR); 

71 busyness = 0..1 edge density. 

72 """ 

73 small = cv2.resize( 

74 img, (img.shape[1] // SEAM_SCORE_SCALE, img.shape[0] // SEAM_SCORE_SCALE), interpolation=cv2.INTER_AREA 

75 ) 

76 v = cv2.resize(_valid_mask_calc(img, dark=False), (small.shape[1], small.shape[0]), interpolation=cv2.INTER_NEAREST) 

77 centr = cv2.distanceTransform((v > 0).astype(np.uint8), cv2.DIST_L2, 3) * SEAM_SCORE_SCALE 

78 grey = cv2.cvtColor(small, cv2.COLOR_BGR2GRAY).astype(np.float32) 

79 lit = np.clip( 

80 (cv2.blur(grey, (SEAM_LIT_BLUR, SEAM_LIT_BLUR)) - SEAM_DARK) / (SEAM_LIT - SEAM_DARK), SEAM_DARK_FLOOR, 1.0 

81 ) 

82 gx, gy = cv2.Sobel(grey, cv2.CV_32F, 1, 0), cv2.Sobel(grey, cv2.CV_32F, 0, 1) 

83 busy = np.clip(cv2.blur(cv2.magnitude(gx, gy), (5, 5)) / 120.0, 0, 1) 

84 return (centr * lit).astype(np.float32), busy.astype(np.float32)

100class _Grid: 

101 """The ownership grid over the atlas (or a region of it) and each screenshot's data on it. 

102 

103 Cells are SEAM_GRID atlas units wide. 

104 

105 Attributes: 

106 atlas: The Atlas. 

107 frames: The placed screenshot records on the grid (labels are indices into this list). 

108 x0: Atlas x of the grid's left edge. 

109 y0: Atlas y of the grid's top edge. 

110 GW: Grid width in cells. 

111 GH: Grid height in cells. 

112 Tg: 3x3 map atlas units -> grid cells. 

113 shot: Screenshot index -> (gx0, gy0, score, busy) on the grid (float16), or None when off the grid; 

114 filled by load. 

115 centre: (n, 2) atlas position of each screenshot's centre. 

116 """ 

117 

118 def __init__(self, atlas, frames, x0, y0, GW, GH): 

119 """Set up an empty grid. 

120 

121 Args: 

122 atlas: The Atlas. 

123 frames: Placed screenshot records to use. 

124 x0: Atlas x of the grid's left edge. 

125 y0: Atlas y of the grid's top edge. 

126 GW: Grid width in cells. 

127 GH: Grid height in cells. 

128 """ 

129 self.atlas, self.frames = atlas, frames 

130 self.x0, self.y0, self.GW, self.GH = x0, y0, GW, GH 

131 self.Tg = np.array([[1 / SEAM_GRID, 0, -x0 / SEAM_GRID], [0, 1 / SEAM_GRID, -y0 / SEAM_GRID], [0, 0, 1]], float) 

132 self.shot: dict[int, tuple] = {} # i -> (gx0, gy0, score, busy) on the grid (float16) 

133 self.centre = np.array([persp(_H(atlas, f), [[f["w"] / 2, f["h"] / 2]])[0] for f in frames], float).reshape( 

134 -1, 2 

135 )

137 def window(self, i): 

138 """Return the grid cells screenshot i covers. 

139 

140 Args: 

141 i: Screenshot index into frames. 

142 

143 Returns: 

144 (gx0, gy0, w, h, Hg): the window's corner and size in cells and the 3x3 map screen px -> grid cells, 

145 or None when the screenshot lies off the grid. 

146 """ 

147 f = self.frames[i] 

148 Hg = self.Tg @ _H(self.atlas, f) 

149 c = persp(Hg, [[0, 0], [f["w"], 0], [f["w"], f["h"]], [0, f["h"]]]) 

150 gx0, gy0 = np.maximum(np.floor(c.min(axis=0)).astype(int) - 1, 0) 

151 gx1, gy1 = np.minimum(np.ceil(c.max(axis=0)).astype(int) + 2, [self.GW, self.GH]) 

152 if gx1 <= gx0 or gy1 <= gy0: 

153 return None 

154 return int(gx0), int(gy0), int(gx1 - gx0), int(gy1 - gy0), Hg

156 def load(self, i): 

157 """Read screenshot i and keep its score / busyness on the grid. 

158 

159 The score is a distance, so it is sampled (no averaging); the busyness is averaged over the cell. 

160 

161 Args: 

162 i: Screenshot index into frames. 

163 

164 Returns: 

165 (gx0, gy0, score, busy) as stored in shot[i], or None when the screenshot lies off the grid. 

166 """ 

167 win = self.window(i) 

168 if win is None: 

169 self.shot[i] = None 

170 return None 

171 gx0, gy0, w, h, Hg = win 

172 img = self.atlas.image(self.frames[i]["path"]) 

173 if img is None: # deleted since Fix seams began, with no copy: it owns nothing (user: aggressive checking) 

174 self.shot[i] = None 

175 return None 

176 sc, busy = _score(img) 

177 S = np.diag([float(SEAM_SCORE_SCALE), float(SEAM_SCORE_SCALE), 1.0]) 

178 Hroi = np.array([[1, 0, -gx0], [0, 1, -gy0], [0, 0, 1]], float) @ Hg @ S 

179 # the score is a distance: sample it (no averaging), the busyness averaged over the cell 

180 s = cv2.warpPerspective(sc, Hroi, (w, h), flags=cv2.INTER_LINEAR, borderValue=0) 

181 b = cv2.warpPerspective(busy, Hroi, (w, h), flags=cv2.INTER_AREA, borderValue=0) 

182 self.shot[i] = (gx0, gy0, s.astype(np.float16), b.astype(np.float16)) 

183 return self.shot[i]

185 def off_middle(self, i, gx0, gy0, w, h): 

186 """Return 0 (screen middle) .. 1 (corner) for each cell of screenshot i's window. 

187 

188 See _off_middle; positions are converted to a 1920x1080 screen. 

189 

190 Args: 

191 i: Screenshot index into frames. 

192 gx0: Window's left cell. 

193 gy0: Window's top cell. 

194 w: Window width in cells. 

195 h: Window height in cells. 

196 

197 Returns: 

198 (h, w) float32 array. 

199 """ 

200 f = self.frames[i] 

201 Hs = np.linalg.inv(self.Tg @ _H(self.atlas, f)) 

202 gx, gy = np.meshgrid(np.arange(w, dtype=np.float64) + gx0 + 0.5, np.arange(h, dtype=np.float64) + gy0 + 0.5) 

203 X = Hs[0, 0] * gx + Hs[0, 1] * gy + Hs[0, 2] 

204 Y = Hs[1, 0] * gx + Hs[1, 1] * gy + Hs[1, 2] 

205 Z = Hs[2, 0] * gx + Hs[2, 1] * gy + Hs[2, 2] 

206 return _off_middle(X / Z * (1920 / f["w"]), Y / Z * (1080 / f["h"]))

209def _off_middle(x, y): 

210 """Measure how far a screen spot is from the screenshot's well-seen middle. 

211 

212 0 at its middle, 1 on the edge of MIDDLE (a trapezoid, SEAM_MIDDLE_TOP / SEAM_MIDDLE_BOTTOM: the tilted 

213 camera sees the ground near the screen's bottom steeper), more outside it. 

214 

215 Args: 

216 x: Screen x on a 1920x1080 screen (px), array. 

217 y: Screen y on a 1920x1080 screen (px), same shape. 

218 

219 Returns: 

220 float32 array of the same shape. 

221 """ 

222 (ty, thw), (by, bhw) = SEAM_MIDDLE_TOP, SEAM_MIDDLE_BOTTOM 

223 hw = np.maximum(thw + (bhw - thw) * (y - ty) / (by - ty), 150.0) 

224 my, hh = (ty + by) / 2, (by - ty) / 2 

225 return np.maximum(np.abs(x - 960.0) / hw, np.abs(y - my) / hh).astype(np.float32)

228def _data(s, e): 

229 """Cost of taking each spot from a screenshot: off its middle (e), near its edge / HUD or in fog (s). 

230 

231 Args: 

232 s: Score of the screenshot on the grid (see _score). 

233 e: Off-middle measure on the grid (see _off_middle), same shape. 

234 

235 Returns: 

236 (cost, feasible): float32 cost per cell, and a bool array of the cells the screenshot can give 

237 (score > 0.5). 

238 """ 

239 s = s.astype(np.float32) 

240 return SEAM_DATA_W * np.minimum(e, 1.6) ** 2 + SEAM_EDGE_W * (1.0 - np.minimum(1.0, s / SEAM_S_FULL)), s > 0.5

243def _protect(atlas, grid) -> np.ndarray: 

244 """Extra seam weight per cell: map nodes and icons (whole from one screenshot), paths. 

245 

246 Cores are each map node with its icon row (the hull of the icons and their map node: one piece), plus icons 

247 whose map node wasn't found (the icons still mark a map node under them). Seams must not cut a core; near 

248 one they cost more, falling off with distance, so where they have to pass between crowded map nodes they 

249 keep to the middle of the gap. Uses the atlas-wide result (atlas.final); nothing is protected without it. 

250 

251 Args: 

252 atlas: The Atlas. 

253 grid: The _Grid. 

254 

255 Returns: 

256 (GH, GW) float32 extra weight. 

257 """ 

258 fin = atlas.final or {} 

259 out = np.zeros((grid.GH, grid.GW), np.float32) 

260 g = lambda x, y: (int(round((x - grid.x0) / SEAM_GRID)), int(round((y - grid.y0) / SEAM_GRID))) # noqa: E731 

261 by_id = {n["id"]: n for n in fin.get("nodes", [])} 

262 paths = np.zeros_like(out, np.uint8) 

263 for p in fin.get("paths", []): 

264 a, b = by_id.get(p["a"]), by_id.get(p["b"]) 

265 if a is not None and b is not None: 

266 cv2.line(paths, g(a["x"], a["y"]), g(b["x"], b["y"]), 1, max(1, round(2 * SEAM_PATH_HALF / SEAM_GRID))) 

267 # cores: each map node with its icon row (the hull of the icons and their map node: one piece), plus 

268 # icons whose map node wasn't found (the icons still mark a map node under them) 

269 core = np.zeros_like(out, np.uint8) 

270 for n in fin.get("nodes", []): 

271 if n["status"] != "removed": 

272 cv2.circle(core, g(n["x"], n["y"]), round(SEAM_NODE_R / SEAM_GRID), 1, -1) 

273 rows: dict[tuple, list] = {} 

274 for ic in fin.get("icons", []): 

275 bx, by = ic.get("bx", ic["x"]), ic.get("by", ic["y"]) 

276 rows.setdefault((round(bx / 20), round(by / 20)), []).append((ic["x"], ic["y"], bx, by)) 

277 for row in rows.values(): 

278 pts = [] 

279 for x, y, bx, by in row: 

280 r = SEAM_ICON_R 

281 pts += [(x - r, y - r), (x + r, y - r), (x - r, y + r), (x + r, y + r), (bx - r, by + r), (bx + r, by + r)] 

282 hull = cv2.convexHull(np.array([g(x, y) for x, y in pts], np.int32)) 

283 cv2.fillConvexPoly(core, hull, 1) 

284 # seams must not cut a core; near one they cost more, falling off with distance, so where they have to 

285 # pass between crowded map nodes they keep to the middle of the gap 

286 d = cv2.distanceTransform((core == 0).astype(np.uint8), cv2.DIST_L2, 3) * SEAM_GRID 

287 near = SEAM_NODE_W * SEAM_NEAR_SHARE * np.clip(1.0 - d / SEAM_NEAR_FALL, 0, 1) 

288 out += SEAM_PATH_W * paths + np.where(core > 0, SEAM_NODE_W, near) 

289 return out

292def _pq(a, axis): 

293 """Return both sides (p, q) of every neighbour pair along an axis, flattened. 

294 

295 Args: 

296 a: 2-D array. 

297 axis: 0 for vertical neighbours, 1 for horizontal neighbours. 

298 

299 Returns: 

300 (p, q) 1-D arrays. 

301 """ 

302 if axis == 1: 

303 return a[:, :-1].ravel(), a[:, 1:].ravel() 

304 return a[:-1, :].ravel(), a[1:, :].ravel()

307def _seam(la, ea, cxa, cya, lb, eb, cxb, cyb, w): 

308 """Seam cost between neighbouring cells owned by la / lb (0 where the same, or one side is empty). 

309 

310 The cost grows with how far off the middle both sides are and how far apart the two screenshots' centres are. 

311 

312 Args: 

313 la: Owner (screenshot index, -1 = none) of the first cells. 

314 ea: Off-middle measure of the first cells in their owner. 

315 cxa: Atlas x of the first owner's centre. 

316 cya: Atlas y of the first owner's centre. 

317 lb: Owner of the second cells. 

318 eb: Off-middle measure of the second cells in their owner. 

319 cxb: Atlas x of the second owner's centre. 

320 cyb: Atlas y of the second owner's centre. 

321 w: Seam weight of the pair (see _protect and busyness). 

322 

323 Returns: 

324 Array of seam costs, one per pair. 

325 """ 

326 d = np.hypot(cxa - cxb, cya - cyb) / 1000.0 

327 c = w * (1.0 + SEAM_OFF_W * (ea + eb) + SEAM_SHIFT_W * d) 

328 return np.where((la != lb) & (la >= 0) & (lb >= 0), c, 0.0)

331def _expand(grid, i, L, U, E, Wt): 

332 """One alpha expansion: hand screenshot i every cell where that lowers the total cost. 

333 

334 Solved as a minimum cut over the cells screenshot i could take. L, U and E are updated in place. 

335 

336 Args: 

337 grid: The _Grid (screenshot i must be loaded). 

338 i: Screenshot index. 

339 L: (GH, GW) int32 owner of each cell (-1 = none). 

340 U: (GH, GW) float32 data cost of each cell in its owner. 

341 E: (GH, GW) float32 off-middle measure of each cell in its owner. 

342 Wt: (GH, GW) float32 seam weight per cell. 

343 

344 Returns: 

345 Number of cells moved to screenshot i. 

346 """ 

347 sh = grid.shot.get(i) 

348 if sh is None: 

349 return 0 

350 gx0, gy0, s, _b = sh 

351 h, w = s.shape 

352 win = (slice(gy0, gy0 + h), slice(gx0, gx0 + w)) 

353 Ea = grid.off_middle(i, gx0, gy0, w, h) 

354 Ua, feas = _data(s, Ea) 

355 Lw, Uw, Ew = L[win], U[win], E[win] 

356 var = feas & (Lw != i) 

357 n = int(var.sum()) 

358 if n == 0: 

359 return 0 

360 cen = grid.centre 

361 Cx, Cy = cen[np.maximum(Lw, 0), 0], cen[np.maximum(Lw, 0), 1] 

362 idx = np.full(Lw.shape, -1, np.int64) 

363 idx[var] = np.arange(n) 

364 delta = np.zeros(n) # cost(switch) - cost(keep) per variable 

365 delta += (Ua - Uw)[var] 

366 rows, cols, caps = [], [], [] 

367 ax, ay = cen[i] 

368 for axis in (0, 1): 

369 (lp, lq), (ep, eq), (cxp, cxq), (cyp, cyq) = (_pq(a, axis) for a in (Lw, Ew, Cx, Cy)) 

370 (eap, eaq), (vp, vq) = _pq(Ea, axis), _pq(idx, axis) 

371 wt = np.maximum(*_pq(Wt[win], axis)) 

372 A = _seam(lp, ep, cxp, cyp, lq, eq, cxq, cyq, wt) # both keep 

373 B = _seam(lp, ep, cxp, cyp, np.full_like(lq, i), eaq, ax, ay, wt) # q switches 

374 C = _seam(np.full_like(lp, i), eap, ax, ay, lq, eq, cxq, cyq, wt) # p switches 

375 both, onlyp, onlyq = (vp >= 0) & (vq >= 0), (vp >= 0) & (vq < 0), (vp < 0) & (vq >= 0) 

376 np.add.at(delta, vp[both], (C - A)[both]) 

377 np.add.at(delta, vq[both], -C[both]) 

378 lam = np.maximum(B + C - A, 0)[both] 

379 rows.append(vp[both]) 

380 cols.append(vq[both]) 

381 caps.append(lam) 

382 np.add.at(delta, vp[onlyp], (C - A)[onlyp]) 

383 np.add.at(delta, vq[onlyq], (B - A)[onlyq]) 

384 src, snk = n, n + 1 

385 v = np.arange(n) 

386 pos = delta > 0 

387 rows += [np.full(int(pos.sum()), src), v[~pos]] 

388 cols += [v[pos], np.full(int((~pos).sum()), snk)] 

389 caps += [delta[pos], -delta[~pos]] 

390 r, c = np.concatenate(rows), np.concatenate(cols) 

391 k = np.round(np.concatenate(caps) * SEAM_FLOW_SCALE).astype(np.int64) 

392 keep = k > 0 

393 r, c, k = r[keep], c[keep], np.minimum(k[keep], 2**30).astype(np.int32) 

394 G = csr_matrix((k, (r, c)), shape=(n + 2, n + 2)) 

395 flow = maximum_flow(G, src, snk).flow 

396 R = (G - flow).tocsr() 

397 R.data[R.data < 0] = 0 

398 R.eliminate_zeros() 

399 reach = breadth_first_order(R, src, directed=True, return_predecessors=False) 

400 stay = np.zeros(n + 2, bool) 

401 stay[reach] = True 

402 move = ~stay[:n] 

403 if not move.any(): 

404 return 0 

405 yy, xx = np.nonzero(var) 

406 yy, xx = yy[move], xx[move] 

407 Lw[yy, xx] = i 

408 Uw[yy, xx] = Ua[yy, xx] 

409 Ew[yy, xx] = Ea[yy, xx] 

410 return int(move.sum())

413def clean_composite(atlas, max_pixels: int = 36_000_000, region=None, log=None): 

414 """Build the stitched map with clean seams. 

415 

416 1. every screenshot on the grid, starting from the cheapest one per cell (no seams counted yet); 2. graph 

417 cuts: each screenshot in turn takes over the cells where that makes the whole map cheaper (up to 

418 SEAM_CYCLES rounds); 3. the ownership grid onto the canvas, and the map drawn from it. The weight is each 

419 spot's score in the quick map's units, so later quick adds take over only where more central. 

420 

421 Args: 

422 atlas: The Atlas. 

423 max_pixels: Largest canvas size in pixels; bigger maps are scaled down. 

424 region: (x0, y0, x1, y1) in atlas units builds only that part (tests, previews); None for everything. 

425 log: Optional function taking one progress line. 

426 

427 Returns: 

428 (canvas, (x0, y0, scale), weight, labels, frame names): BGR canvas, its origin, (h, w) float32 weight, 

429 (h, w) int16 owner per canvas pixel (index into frame names, -1 = none), and the screenshot names. 

430 """ 

431 frames = [f for f in atlas.frames if atlas.has_picture(f["path"])] # (the screenshot, else its copy) 

432 if not frames: 

433 z = np.zeros((1, 1), np.float32) 

434 return np.zeros((1, 1, 3), np.uint8), (0, 0, 1.0), z, z.astype(np.int16) - 1, [] 

435 polys = [persp(_H(atlas, f), [[0, 0], [f["w"], 0], [f["w"], f["h"]], [0, f["h"]]]) for f in frames] 

436 if region is not None: 

437 rx0, ry0, rx1, ry1 = region 

438 keep = [ 

439 i 

440 for i, p in enumerate(polys) 

441 if p[:, 0].max() > rx0 and p[:, 0].min() < rx1 and p[:, 1].max() > ry0 and p[:, 1].min() < ry1 

442 ] 

443 frames, polys = [frames[i] for i in keep], [polys[i] for i in keep] 

444 x0, y0, x1, y1 = int(rx0), int(ry0), int(math.ceil(rx1)), int(math.ceil(ry1)) 

445 else: 

446 allc = np.vstack(polys) 

447 x0, y0 = np.floor(allc.min(axis=0)).astype(int) 

448 x1, y1 = np.ceil(allc.max(axis=0)).astype(int) 

449 W, H = int(x1 - x0), int(y1 - y0) 

450 scale = min(1.0, math.sqrt(max_pixels / max(1, W * H))) 

451 CW, CH = max(1, int(W * scale)), max(1, int(H * scale)) 

452 T = np.array([[scale, 0, -x0 * scale], [0, scale, -y0 * scale], [0, 0, 1]], float) 

453 

454 # 1. every screenshot on the grid; start from the cheapest one per cell (no seams counted yet) 

455 GW, GH = math.ceil(W / SEAM_GRID), math.ceil(H / SEAM_GRID) 

456 grid = _Grid(atlas, frames, x0, y0, GW, GH) 

457 L = np.full((GH, GW), -1, np.int32) 

458 U = np.full((GH, GW), np.inf, np.float32) 

459 E = np.zeros((GH, GW), np.float32) 

460 busy = np.zeros((GH, GW), np.float32) 

461 for i in range(len(frames)): 

462 sh = grid.load(i) 

463 if sh is not None: 

464 gx0, gy0, s, b = sh 

465 e = grid.off_middle(i, gx0, gy0, s.shape[1], s.shape[0]) 

466 u, feas = _data(s, e) 

467 win = (slice(gy0, gy0 + s.shape[0]), slice(gx0, gx0 + s.shape[1])) 

468 take = feas & (u < U[win]) 

469 L[win][take], U[win][take], E[win][take], busy[win][take] = i, u[take], e[take], b[take] 

470 if log and i % 100 == 99: 

471 log(f"clean map: read {i + 1} of {len(frames)} screenshots") 

472 U[L < 0] = 0 

473 Wt = 1.0 + SEAM_TEX_W * busy + _protect(atlas, grid) 

474 

475 # 2. graph cuts: each screenshot in turn takes over the cells where that makes the whole map cheaper 

476 order = sorted(range(len(frames)), key=lambda i: -float(np.mean(grid.shot[i][2])) if grid.shot[i] else 0) 

477 for cycle in range(SEAM_CYCLES): 

478 moved = 0 

479 for k, i in enumerate(order): 

480 moved += _expand(grid, i, L, U, E, Wt) 

481 if log and k % 100 == 99: 

482 log(f"clean map: seams, round {cycle + 1} of {SEAM_CYCLES}: {k + 1} of {len(frames)} screenshots") 

483 if moved == 0: 

484 break 

485 

486 # 3. the ownership grid onto the canvas, and the map drawn from it 

487 Mi = np.array([[1 / (scale * SEAM_GRID), 0, 0], [0, 1 / (scale * SEAM_GRID), 0]], np.float64) 

488 labels = cv2.warpAffine( 

489 L.astype(np.float32), Mi, (CW, CH), flags=cv2.INTER_NEAREST | cv2.WARP_INVERSE_MAP, borderValue=-1 

490 ).astype(np.int16) 

491 canvas = np.zeros((CH, CW, 3), np.uint8) 

492 gone = set() 

493 for i, f in enumerate(frames): 

494 roi = _roi(T @ _H(atlas, f), f, CW, CH) 

495 if roi is None: 

496 continue 

497 bx0, by0, size, Hroi = roi 

498 mine = labels[by0 : by0 + size[1], bx0 : bx0 + size[0]] == i 

499 if not mine.any(): 

500 continue 

501 img = atlas.image(f["path"]) 

502 if img is None: # deleted while the clean map was worked out, with no copy: its part stays empty and owned 

503 # by no one (owned and weighted, the black kept later quick adds from filling it: CodeRabbit) 

504 labels[labels == i] = -1 

505 gone.add(i) 

506 continue 

507 pix = cv2.warpPerspective(img, Hroi, size, flags=cv2.INTER_LINEAR, borderValue=0) 

508 canvas[by0 : by0 + size[1], bx0 : bx0 + size[0]][mine] = pix[mine] 

509 if log and i % 100 == 99: 

510 log(f"clean map: drawn {i + 1} of {len(frames)} screenshots") 

511 # each spot's score (the quick map's units: later quick adds take over only where more central) 

512 sgrid = np.zeros((GH, GW), np.float32) 

513 for i, sh in grid.shot.items(): 

514 if sh is not None and i not in gone: 

515 gx0, gy0, s, _b = sh 

516 win = (slice(gy0, gy0 + s.shape[0]), slice(gx0, gx0 + s.shape[1])) 

517 mine = L[win] == i 

518 sgrid[win][mine] = s[mine] 

519 best = cv2.warpAffine(sgrid, Mi, (CW, CH), flags=cv2.INTER_LINEAR | cv2.WARP_INVERSE_MAP, borderValue=0) 

520 best[labels >= 0] = np.maximum(best[labels >= 0], 1.0) 

521 return canvas, (x0, y0, scale), best, labels, [f["name"] for f in frames]

524def _roi(Hc, f, CW, CH): 

525 """Return the canvas box a screenshot lands in. 

526 

527 Args: 

528 Hc: 3x3 map screen px -> canvas px. 

529 f: Placed screenshot record (w, h). 

530 CW: Canvas width in px. 

531 CH: Canvas height in px. 

532 

533 Returns: 

534 (bx0, by0, (w, h), Hroi): box corner, box size and the 3x3 map screen px -> box px; None when the 

535 screenshot misses the canvas. 

536 """ 

537 c = persp(Hc, [[0, 0], [f["w"], 0], [f["w"], f["h"]], [0, f["h"]]]) 

538 bx0, by0 = np.maximum(np.floor(c.min(axis=0)).astype(int), 0) 

539 bx1, by1 = np.minimum(np.ceil(c.max(axis=0)).astype(int), [CW, CH]) 

540 if bx1 <= bx0 or by1 <= by0: 

541 return None 

542 Hroi = np.array([[1, 0, -bx0], [0, 1, -by0], [0, 0, 1]], float) @ Hc 

543 return int(bx0), int(by0), (int(bx1 - bx0), int(by1 - by0)), Hroi
