1"""Safe file writing and the worker-process setup shared by the builder, the atlas-wide check and the viewer. 

2 

3write_text_atomic: the atlas, the settings and the detection caches are rewritten often (every bookmark, 

4every added screenshot). Writing them in place left a cut-off file when the program was closed, crashed or 

5the computer slept mid-write, and a cut-off atlas.json loses every bookmark. The text goes to a temporary 

6file next to the target, which then replaces the target in one step: readers see the old file or the new 

7one, never half of one. 

8 

9process_pool: worker processes are always started fresh ("spawn"), as Windows and macOS do anyway. Linux 

10would otherwise copy ("fork") the viewer process, threads and Tk included, which can hang. 

11"""

24def write_text_atomic(path, text: str, encoding: str = "utf-8") -> None: 

25 """Write a text file so readers see the old file or the new one, never half of one. 

26 

27 The text goes to a temporary file next to the target, which then replaces the target in one step (see the 

28 module docstring). On Windows a virus scanner / cloud sync may hold the target briefly, so the replace is 

29 tried 5 times; the temporary file is removed on any failure. 

30 

31 Args: 

32 path: Target file path. 

33 text: The file contents. 

34 encoding: Text encoding. 

35 

36 Raises: 

37 PermissionError: The target stayed locked through every try. 

38 """ 

39 path = Path(path) 

40 fd, tmp = tempfile.mkstemp(prefix=f".{path.name}.", suffix=".tmp", dir=path.parent) 

41 try: 

42 with os.fdopen(fd, "w", encoding=encoding, newline="") as f: 

43 f.write(text) 

44 f.flush() 

45 os.fsync(f.fileno()) 

46 for attempt in range(5): # Windows: a virus scanner / cloud sync may hold the target briefly 

47 try: 

48 os.replace(tmp, path) 

49 return 

50 except PermissionError: 

51 if attempt == 4: 

52 raise 

53 time.sleep(0.1 * (attempt + 1)) 

54 except BaseException: 

55 with contextlib.suppress(OSError): 

56 os.unlink(tmp) 

57 raise

60def read_json(path) -> dict | None: 

61 """Read the JSON in a file. 

62 

63 Args: 

64 path: File path. 

65 

66 Returns: 

67 The parsed JSON, or None when it is missing, cut off or not JSON (e.g. a cache from a crash). 

68 """ 

69 try: 

70 return json.loads(Path(path).read_text(encoding="utf-8")) 

71 except (OSError, ValueError): 

72 return None

75def workers(wanted: int) -> int: 

76 """Choose how many worker processes to use: at most `wanted`, and one core left for the window. 

77 

78 Args: 

79 wanted: Number of workers wanted. 

80 

81 Returns: 

82 Number of workers, at least 1. 

83 """ 

84 return max(1, min(wanted, (os.cpu_count() or 2) - 1))

87def process_pool(max_workers: int): 

88 """Create a process pool whose workers are started fresh ("spawn"; see the module docstring). 

89 

90 Args: 

91 max_workers: Number of worker processes. 

92 

93 Returns: 

94 A concurrent.futures.ProcessPoolExecutor.

96 from concurrent.futures import ProcessPoolExecutor 

97 

98 return ProcessPoolExecutor(max_workers=max_workers, mp_context=multiprocessing.get_context("spawn"))
