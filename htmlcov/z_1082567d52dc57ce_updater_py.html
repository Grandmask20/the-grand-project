1"""app.updater: part of AtlasSession (a mixin; see app/session.py): the in-app update, as Path of Building has it. 

2 

3The player release (channel "release", with manifest.json: grand_project/release_manifest.py) asks GitHub for the 

4latest release once a day when the app starts (a setting turns it off) and whenever the status bar's version is 

5pressed. A newer one is shown with its notes; Update now downloads its zip into user_data/update/, checks every 

6file against its manifest, works out which files change (and whether the Python packages must be installed again), 

7writes the plan, starts app/update_apply.py on its own and quits. That helper puts the files in place once the app has 

8closed, with a backup it puts back on any failure, and starts the app again; the page reloads itself on the new 

9version. Development copies (internal, develop: no manifest) are updated with git instead. 

10 

11Only the release's own files (its manifest's) change: atlases/, user_data/, local_paths.json and .venv/ never do. 

12"""

38def latest_release(fetch=None) -> dict: 

39 """Ask GitHub for the latest release. 

40 

41 Args: 

42 fetch: url -> parsed JSON (default: an HTTPS GET with the app's name as User-Agent; tests pass their own). 

43 

44 Returns: 

45 {"release", "name", "notes", "url", "zip", "published"}. 

46 

47 Raises: 

48 OSError / ValueError: When GitHub can't be reached or answers something else. 

49 """ 

50 data = (fetch or _get_json)(K.UPDATE_RELEASES_URL) 

51 tag = str(data.get("tag_name", "")) 

52 if not rm.release_key(tag) or not data.get("zipball_url"): 

53 raise ValueError(f"GitHub's latest release has no release number ({tag!r})") 

54 return { 

55 "release": tag.lstrip("vV"), 

56 "name": data.get("name") or tag, 

57 "notes": data.get("body") or "", 

58 "url": data.get("html_url") or "", 

59 "zip": data["zipball_url"], 

60 "published": data.get("published_at") or "", 

61 }

64def _get_json(url: str) -> dict: 

65 req = urllib.request.Request(url, headers={"User-Agent": K.UPDATE_AGENT, "Accept": "application/vnd.github+json"}) 

66 try: 

67 with urllib.request.urlopen(req, timeout=K.UPDATE_HTTP_WAIT) as r: # noqa: S310 (a fixed https URL) 

68 return json.loads(r.read().decode("utf-8")) 

69 except urllib.error.HTTPError as e: 

70 raise OSError(limit_message(e.code, e.headers) or f"GitHub answered {e.code}") from e 

71 except http.client.HTTPException as e: # (an answer cut off midway: it isn't an OSError, CodeRabbit) 

72 raise OSError(f"GitHub's answer was cut off ({e})") from e

75def limit_message(code: int, headers) -> str | None: 

76 """What to say when GitHub's limit of checks is used up (403 / 429 with no checks left), else None. 

77 

78 Args: 

79 code: The HTTP status. 

80 headers: The answer's headers (X-RateLimit-Remaining, X-RateLimit-Reset: when it starts again, s since 1970). 

81 """ 

82 if code not in (403, 429) or str(headers.get("X-RateLimit-Remaining", "")) != "0": 

83 return None 

84 reset = headers.get("X-RateLimit-Reset") 

85 at = time.strftime("%H:%M", time.localtime(int(reset))) if str(reset or "").isdigit() else "within the hour" 

86 return f"GitHub's limit of update checks from your address is used up: it answers again at {at}"

89def unpack(zip_path: Path, dest: Path) -> dict: 

90 """Unpack a release zip (GitHub's: one top folder) and check its files against its own manifest. 

91 

92 Args: 

93 zip_path: The zip. 

94 dest: An empty folder for the release's files. 

95 

96 Returns: 

97 The release's manifest. 

98 

99 Raises: 

100 ValueError: When it isn't a release (no manifest), a file is missing or differs from its hash, or a path in 

101 it would land outside `dest`. 

102 """ 

103 with zipfile.ZipFile(zip_path) as z: 

104 names = z.namelist() 

105 tops = {n.split("/", 1)[0] for n in names if n} 

106 top = tops.pop() + "/" if len(tops) == 1 else "" 

107 try: 

108 manifest = json.loads(z.read(top + rm.MANIFEST).decode("utf-8")) 

109 except (KeyError, ValueError) as e: 

110 raise ValueError("the download has no release manifest: not a release") from e 

111 files = manifest.get("files") if isinstance(manifest, dict) else None 

112 if not isinstance(files, dict) or not manifest.get("release"): 

113 raise ValueError("the download's release manifest is broken") 

114 root = dest.resolve() 

115 for rel, want in files.items(): 

116 out = (dest / rel).resolve() 

117 if root not in out.parents: 

118 raise ValueError(f"the download has a file outside the program folder: {rel}") 

119 try: 

120 data = z.read(top + rel) 

121 except KeyError as e: 

122 raise ValueError(f"the download lacks {rel}") from e 

123 if rm.content_hash(data) != want: 

124 raise ValueError(f"{rel} in the download isn't the release's (broken download?)") 

125 out.parent.mkdir(parents=True, exist_ok=True) 

126 out.write_bytes(data) 

127 # its permissions as the release has them (start.sh can be run: it lost them, CodeRabbit) 

128 mode = (z.getinfo(top + rel).external_attr >> 16) & 0o777 

129 if mode and os.name != "nt": 

130 out.chmod(mode) 

131 (dest / rm.MANIFEST).write_text(json.dumps(manifest, indent=1), encoding="utf-8") 

132 return manifest

135def plan_files(root: Path, installed: dict, new: dict) -> tuple[list[str], list[str], bool]: 

136 """Which files an update changes. 

137 

138 Args: 

139 root: The program folder. 

140 installed: Its manifest. 

141 new: The new release's manifest. 

142 

143 Returns: 

144 (files to copy in: new or changed, files to remove: the installed release's that the new one dropped, 

145 whether requirements.txt changes). 

146 """ 

147 copy = [f for f, h in new["files"].items() if rm.file_hash(root / f) != h] 

148 copy.append(rm.MANIFEST) 

149 drop = [f for f in installed.get("files", {}) if f not in new["files"] and (root / f).is_file()] 

150 return copy, drop, "requirements.txt" in copy

167 def _update_state(self) -> dict: 

168 if not hasattr(self, "_update"): 

169 self._update = { 

170 "state": "idle", 

171 "latest": None, 

172 "error": None, 

173 "got": 0, 

174 "size": 0, 

175 "just": None, 

176 "seq": 0, 

177 # when this run of the app started: the numbers start again with each run, so the page orders 

178 # statuses by (run, seq) (a restarted app's were dropped as older than the last run's: CodeRabbit) 

179 "run": round(time.time(), 3), 

180 } 

181 self._update_lock = threading.Lock() 

182 return self._update

184 def update_status(self) -> dict: 

185 """What the page shows for the update (the status bar's version and its box). 

186 

187 Returns: 

188 {"channel", "release", "program", "can", "why", "auto", "checked", "next", "state", "latest", 

189 "available", "error", "got", "size", "just", "seq", "run"}; state is idle / checking / downloading / 

190 restarting / 

191 error; next: when GitHub may be asked again (s since 1970; 0: now), a time and not a count of seconds, 

192 so a page that opens the box later counts down to the same moment (user: it started again at 60). 

193 """ 

194 u = self._update_state() 

195 seq = u["seq"] # (first: a change made meanwhile comes with a higher number, in its own event) 

196 installed = rm.read(rc_paths.ROOT) 

197 latest = u["latest"] 

198 mine = installed["release"] if installed else None 

199 newer = bool(latest and mine and rm.release_key(latest["release"]) > rm.release_key(mine)) 

200 can, why = self._can_update(installed) 

201 return { 

202 "channel": rc_paths.CHANNEL, 

203 "release": mine, 

204 "program": K.VERSION, 

205 "can": can, 

206 "why": why, 

207 "auto": bool(self.settings.get("update_auto", True)), 

208 "checked": self.settings.get("update_checked"), 

209 "next": self._next_check(), 

210 "state": u["state"], 

211 "latest": latest, 

212 "available": newer, 

213 "error": u["error"], 

214 "got": u["got"], 

215 "size": u["size"], 

216 "just": u["just"], 

217 # goes up with each change: the page keeps the newest it has (a button's answer could land after the 

218 # event of what followed, and showed "Checking…" for good: CodeRabbit) 

219 "seq": seq, 

220 "run": u["run"], 

221 }

223 def _next_check(self) -> float: 

224 left = self._update_state().get("asked", -1e9) + K.UPDATE_CHECK_GAP - time.monotonic() 

225 return round(time.time() + left, 1) if left > 0 else 0

227 def _can_update(self, installed: dict | None) -> tuple[bool, str]: 

228 if rc_paths.CHANNEL != "release" or installed is None: 

229 return False, "This is a development copy: update it with git (git pull)." 

230 if (rc_paths.ROOT / ".git").exists(): 

231 return False, "This copy is a git clone: update it with git pull." 

232 return True, ""

245 def check_update(self, background: bool = True, fetch=None) -> None: 

246 """Ask GitHub for the latest release (once at a time, and at most once per UPDATE_CHECK_GAP: in between the 

247 last answer stands). 

248 

249 Args: 

250 background: On a thread of its own (the page's button); False runs it here (tests). 

251 fetch: As latest_release's. 

252 """ 

253 u = self._update_state() 

254 with self._update_lock: 

255 if u["state"] in ("checking", "downloading", "restarting"): 

256 return 

257 if time.monotonic() - u.get("asked", -1e9) < K.UPDATE_CHECK_GAP: 

258 self._publish_update() # (the page learns how long to wait) 

259 return 

260 u.update(state="checking", error=None, asked=time.monotonic()) 

261 self._publish_update() 

262 

263 def run() -> None: 

264 try: 

265 latest = latest_release(fetch) 

266 u.update(state="idle", latest=latest) 

267 self.set_setting("update_checked", time.time()) 

268 st = self.update_status() 

269 self.events.log( 

270 f"Update check: {latest['name']} is out (you have {st['release']})" 

271 if st["available"] 

272 else f"Update check: you have the latest release ({st['release'] or K.VERSION})" 

273 ) 

274 except Exception as e: # noqa: BLE001 (any failure ends the check: it stayed "checking" for good) 

275 u.update(state="error", error=f"Couldn't reach GitHub to check for an update: {e}") 

276 LOG.warning(f"update check failed: {e}") 

277 self._publish_update() 

278 

279 if background: 

280 threading.Thread(target=run, name="update-check", daemon=True).start() 

281 else: 

282 run()

284 def check_update_at_start(self, fetch=None) -> bool: 

285 """The once-a-day check when the app starts (the player release, when the setting is on). 

286 

287 Returns: 

288 True when a check was started. 

289 """ 

290 st = self.update_status() 

291 last = st["checked"] or 0 

292 if not st["can"] or not st["auto"] or time.time() - float(last) < K.UPDATE_CHECK_EVERY: 

293 return False 

294 self.check_update(fetch=fetch) 

295 return True

297 def install_update( 

298 self, quit_app, download=None, launch=None, background: bool = True, restart_args: list[str] | None = None 

299 ) -> None: 

300 """Download the latest release, check it, and hand over to the helper that puts it in place (the app quits). 

301 

302 Args: 

303 quit_app: Stops the app (saves the atlas and settings first). 

304 download: (url, file, on_progress(got, size)) -> None (default: an HTTPS download; tests pass their own). 

305 launch: (argv) -> None, starts the helper (default: a process of its own). 

306 background: On a thread of its own; False runs it here (tests). 

307 restart_args: More arguments for the app's restart (its port: the page reconnects to the same address, 

308 and a copy started on the default port could find another one running there). 

309 

310 Raises: 

311 ValueError: When this copy can't be updated in the app, or there is nothing newer. 

312 """ 

313 st = self.update_status() 

314 if not st["can"]: 

315 raise ValueError(st["why"]) 

316 if not st["available"]: 

317 raise ValueError("There is no newer release to install: check for an update first.") 

318 u = self._update_state() 

319 with self._update_lock: 

320 if u["state"] in ("checking", "downloading", "restarting"): 

321 raise ValueError("An update check or download is already running.") 

322 u.update(state="downloading", error=None, got=0, size=0) 

323 latest, mine = st["latest"], st["release"] 

324 self.events.log(f"Updating to {latest['name']}: downloading it") 

325 self._publish_update() 

326 

327 def progress(got: int, size: int) -> None: 

328 u.update(got=got, size=size) 

329 self._publish_update()

331 def run() -> None: 

332 work = rc_paths.USER_DATA / "update" 

333 try: 

334 shutil.rmtree(work, ignore_errors=True) # (an earlier, unfinished one) 

335 work.mkdir(parents=True, exist_ok=True) 

336 zip_path = work / "release.zip" 

337 (download or _download)(latest["zip"], zip_path, progress) 

338 new = unpack(zip_path, work / "new") 

339 if rm.release_key(new["release"]) <= rm.release_key(mine or ""): 

340 raise ValueError(f"the download is release {new['release']}, not newer than {mine}") 

341 installed = rm.read(rc_paths.ROOT) or {"files": {}} 

342 copy, drop, pip = plan_files(rc_paths.ROOT, installed, new) 

343 helper = work / "update_apply.py" # (run from a copy: the program's own changes under it) 

344 shutil.copy2(Path(__file__).with_name("update_apply.py"), helper) 

345 plan = { 

346 "pid": os.getpid(), 

347 "root": str(rc_paths.ROOT), 

348 "new": str(work / "new"), 

349 "backup": str(work / "backup"), 

350 "copy": copy, 

351 "delete": drop, 

352 "pip": pip, 

353 "python": console_python(), 

354 "restart": [sys.executable, "-m", "grand_project", "--no-browser", *(restart_args or [])], 

355 "result": str(rc_paths.USER_DATA / K.UPDATE_RESULT), 

356 "from": mine, 

357 "to": new["release"], 

358 } 

359 (work / "plan.json").write_text(json.dumps(plan, indent=1), encoding="utf-8") 

360 self.events.log( 

361 f"Updating to {latest['name']}: {len(copy)} files to replace, {len(drop)} to remove" 

362 + (", Python packages to install" if pip else "") 

363 + "; the app restarts" 

364 ) 

365 u.update(state="restarting") 

366 self._publish_update() 

367 (launch or _launch)([sys.executable, "-I", str(helper), str(work / "plan.json")]) 

368 quit_app() 

369 except Exception as e: # noqa: BLE001 (any failure: said, nothing changed) 

370 u.update(state="error", error=f"The update couldn't be prepared: {e}") 

371 LOG.warning(f"update failed: {e}") 

372 self._publish_update() 

373 

374 if background: 

375 threading.Thread(target=run, name="update-install", daemon=True).start() 

376 else: 

377 run()

379 def after_update(self) -> dict | None: 

380 """At start: what the last update did (the helper's result), said in the log and to the page once. 

381 

382 Returns: 

383 The result, or None when there was none. 

384 """ 

385 f = rc_paths.USER_DATA / K.UPDATE_RESULT 

386 try: 

387 r = json.loads(f.read_text(encoding="utf-8")) 

388 except (OSError, ValueError): 

389 return None 

390 f.unlink(missing_ok=True) 

391 if r.get("ok"): 

392 self.events.log(f"Updated from release {r.get('from')} to {r.get('to')}") 

393 shutil.rmtree(rc_paths.USER_DATA / "update", ignore_errors=True) 

394 else: 

395 self.events.log(f"The update to release {r.get('to')} failed, nothing changed: {r.get('error')}") 

396 self._update_state()["just"] = r 

397 return r

400def _download(url: str, dest: Path, progress) -> None: 

401 req = urllib.request.Request(url, headers={"User-Agent": K.UPDATE_AGENT}) 

402 with urllib.request.urlopen(req, timeout=K.UPDATE_HTTP_WAIT) as r, dest.open("wb") as f: # noqa: S310 

403 size = int(r.headers.get("Content-Length") or 0) 

404 got, last = 0, 0.0 

405 while chunk := r.read(K.UPDATE_CHUNK): 

406 f.write(chunk) 

407 got += len(chunk) 

408 if time.monotonic() - last > K.UPDATE_PROGRESS_EVERY: 

409 last = time.monotonic() 

410 progress(got, size) 

411 progress(got, size)
